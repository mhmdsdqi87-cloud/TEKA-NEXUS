   386→// ─── Global search dialog ────────────────────────────────────
   387→export function SearchDialog() {
   388→  const open = useApp((s) => s.searchOpen);
   389→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   390→  const navigate = useApp((s) => s.navigate);
   391→
   392→  const [query, setQuery] = useState("");
   393→  const [data, setData] = useState<{ q: string; results: SearchResults } | null>(null);
   394→  const reqId = useRef(0);
   395→
   396→  // Debounced server search (250ms, min 2 chars).
   397→  useEffect(() => {
   398→    const q = query.trim();
   399→    if (q.length < 2) return;
   400→    const id = ++reqId.current;
   401→    const timer = setTimeout(async () => {
   402→      try {
   403→        const res = await get<SearchResults>(`/api/search${qs({ q })}`);
   404→        if (reqId.current === id) setData({ q, results: res });
   405→      } catch {
   406→        if (reqId.current === id) setData({ q, results: { projects: [], users: [], tags: [] } });
   407→      }
   408→    }, 250);
   409→    return () => clearTimeout(timer);
   410→  }, [query]);
   411→
   412→  // Close + reset (event handler, not an effect).
   413→  const handleOpenChange = (next: boolean) => {
   414→    setSearchOpen(next);
   415→    if (!next) {
   416→      reqId.current++;
   417→      setQuery("");
   418→      setData(null);
   419→    }
   420→  };
   421→
   422→  const q = query.trim();
   423→  // Results are only shown when they belong to the current query;
   424→  // a newer query with no response yet renders the "Searching…" state.
   425→  const results = data && data.q === q ? data.results : null;
   426→  const searching = q.length >= 2 && results === null;
   427→  const hasResults =
   428→    results && (results.projects.length > 0 || results.users.length > 0 || results.tags.length > 0);
   429→
   430→  return (
   431→    <CommandDialog
   432→      open={open}
   433→      onOpenChange={handleOpenChange}
   434→      title="Search TEKA NEXUS"
   435→      description="Search projects, people and tags"
   436→      showCloseButton={false}
   437→      className="max-w-xl top-[15%] translate-y-0"
   438→    >
   439→      <CommandInput
   440→        value={query}
   441→        onValueChange={setQuery}
   442→        placeholder="Search projects, people, tags…"
   443→        aria-label="Search query"
   444→      />
   445→      <CommandList>
   446→        {q.length < 2 ? (
   447→          <div className="py-8 px-4 text-center text-sm text-muted-foreground">
   448→            Type at least 2 characters to search.
   449→          </div>
   450→        ) : searching ? (
   451→          <div className="py-8 px-4 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   452→            <Loader2 className="h-4 w-4 animate-spin" /> Searching…
   453→          </div>
   454→        ) : hasResults ? (
   455→          <>
   456→            {results!.projects.length > 0 && (
   457→              <CommandGroup heading="Projects">
   458→                {results!.projects.map((p) => (
   459→                  <CommandItem
   460→                    key={p.id}
   461→                    value={`${q} ${p.name}`}
   462→                    onSelect={() => {
   463→                      setSearchOpen(false);
   464→                      navigate({ name: "project", slug: p.slug });
   465→                    }}
   466→                  >
   467→                    <FileCode2 className="text-cyan-300" />
   468→                    <span className="truncate">{p.name}</span>
   469→                    <span className="ml-auto truncate max-w-[45%] text-xs text-muted-foreground">{p.tagline}</span>
   470→                  </CommandItem>
   471→                ))}
   472→              </CommandGroup>
   473→            )}
   474→            {results!.users.length > 0 && (
   475→              <CommandGroup heading="Users">
   476→                {results!.users.map((u) => (
   477→                  <CommandItem
   478→                    key={u.id}
   479→                    value={`${q} ${u.displayName} ${u.username}`}
   480→                    onSelect={() => {
   481→                      setSearchOpen(false);
   482→                      navigate({ name: "profile", username: u.username });
   483→                    }}
   484→                  >
   485→                    <UserAvatar user={u} size={20} />
   486→                    <span className="truncate">{u.displayName}</span>
   487→                    <span className="ml-auto text-xs text-muted-foreground">@{u.username}</span>
   488→                  </CommandItem>
   489→                ))}
   490→              </CommandGroup>
   491→            )}
   492→            {results!.tags.length > 0 && (
   493→              <CommandGroup heading="Tags">
   494→                {results!.tags.map((t) => (
   495→                  <CommandItem
   496→                    key={t.tag}
   497→                    value={`${q} ${t.tag}`}
   498→                    onSelect={() => {
   499→                      setSearchOpen(false);
   500→                      navigate({ name: "explore", query: t.tag });
   501→                    }}
   502→                  >
   503→                    <Hash className="text-violet-300" />
   504→                    <span className="font-mono">#{t.tag}</span>
   505→                    <span className="ml-auto text-xs text-muted-foreground">{t.count} project{t.count === 1 ? "" : "s"}</span>
   506→                  </CommandItem>
   507→                ))}
   508→              </CommandGroup>
   509→            )}
   510→          </>
   511→        ) : (
   512→          <CommandEmpty>No results for &ldquo;{q}&rdquo;</CommandEmpty>
   513→        )}
   514→      </CommandList>
   515→    </CommandDialog>
   516→  );
   517→}

   664→export function AppShell({ children }: { children: React.ReactNode }) {
   665→  const user = useApp((s) => s.user);
   666→  const setUnreadCount = useApp((s) => s.setUnreadCount);
   667→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   668→  const userId = user?.id ?? null;
   669→
   670→  // Poll unread notification count every 45s while signed in.
   671→  useEffect(() => {
   672→    if (!userId) return;
   673→    let cancelled = false;
   674→    const poll = async () => {
   675→      try {
   676→        const res = await get<{ notifications: unknown[]; unread: number }>("/api/notifications");
   677→        if (!cancelled) setUnreadCount(res.unread);
   678→      } catch {
   679→        /* silent — background badge poll */
   680→      }
   681→    };
   682→    poll();
   683→    const timer = setInterval(poll, 45_000);
   684→    return () => {
   685→      cancelled = true;
   686→      clearInterval(timer);
   687→    };
   688→  }, [userId, setUnreadCount]);
   689→
   690→  // "/" opens global search (ignored while typing in form fields).
   691→  useEffect(() => {
   692→    const onKey = (e: KeyboardEvent) => {
   693→      if (e.key !== "/") return;
   694→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   695→      const target = e.target as HTMLElement | null;
   696→      const tag = target?.tagName;
   697→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || target?.isContentEditable) return;
   698→      e.preventDefault();
   699→      setSearchOpen(true);
   700→    };
   701→    window.addEventListener("keydown", onKey);
   702→    return () => window.removeEventListener("keydown", onKey);
   703→  }, [setSearchOpen]);
   704→
   705→  return (
   706→    <MotionConfig reducedMotion="user">
   707→      <div className="min-h-screen flex">
   708→        <LeftNav />
   709→
   710→        <div className="flex-1 flex flex-col min-w-0">
   711→          <TopBar />
   712→          <main className="flex-1 pb-24 md:pb-10" id="main-content">
   713→            <div className="max-w-5xl mx-auto px-4 md:px-8 py-6">
   714→              <motion.div
   715→                initial={{ opacity: 0, y: 8 }}
   716→                animate={{ opacity: 1, y: 0 }}
   717→                transition={{ duration: 0.3, ease: "easeOut" }}
   718→              >
   719→                {children}
   720→              </motion.div>
   721→            </div>
   722→          </main>
   723→          <footer className="py-4 text-center text-xs text-muted-foreground/70 select-none" aria-label="Footer">
   724→            TEKA NEXUS — demo preview
   725→          </footer>
   726→        </div>
   727→
   728→        <RightRail />
   729→        <MobileNav />
   730→        <SearchDialog />
   731→      </div>
   732→    </MotionConfig>
   733→  );
   734→}
