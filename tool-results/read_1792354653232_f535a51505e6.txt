   387→   387→export function SearchDialog() {
   388→   388→  const open = useApp((s) => s.searchOpen);
   389→   389→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   390→   390→  const navigate = useApp((s) => s.navigate);
   391→   391→
   392→   392→  const [query, setQuery] = useState("");
   393→   393→  const [data, setData] = useState<{ q: string; results: SearchResults } | null>(null);
   394→   394→  const reqId = useRef(0);
   395→   395→
   396→   396→  // Debounced server search (250ms, min 2 chars).
   397→   397→  useEffect(() => {
   398→   398→    const q = query.trim();
   399→   399→    if (q.length < 2) return;
   400→   400→    const id = ++reqId.current;
   401→   401→    const timer = setTimeout(async () => {
   402→   402→      try {
   403→   403→        const res = await get<SearchResults>(`/api/search${qs({ q })}`);
   404→   404→        if (reqId.current === id) setData({ q, results: res });
   405→   405→      } catch {
   406→   406→        if (reqId.current === id) setData({ q, results: { projects: [], users: [], tags: [] } });
   407→   407→      }
   408→   408→    }, 250);
   409→   409→    return () => clearTimeout(timer);
   410→   410→  }, [query]);
   411→   411→
   412→   412→  // Close + reset (event handler, not an effect).
   413→   413→  const handleOpenChange = (next: boolean) => {
   414→   414→    setSearchOpen(next);
   415→   415→    if (!next) {
   416→   416→      reqId.current++;
   417→   417→      setQuery("");
   418→   418→      setData(null);
   419→   419→    }
   420→   420→  };
   421→   421→
   422→   422→  const q = query.trim();
   423→   423→  // Results are only shown when they belong to the current query;
   424→   424→  // a newer query with no response yet renders the "Searching…" state.
   425→   425→  const results = data && data.q === q ? data.results : null;
   426→   426→  const searching = q.length >= 2 && results === null;
   427→   427→  const hasResults =
   428→   428→    results && (results.projects.length > 0 || results.users.length > 0 || results.tags.length > 0);
   429→   429→
   430→   430→  return (
   431→   431→    <CommandDialog
   432→   432→      open={open}
   433→   433→      onOpenChange={handleOpenChange}
   434→   434→      title="Search TEKA NEXUS"
   435→   435→      description="Search projects, people and tags"
   436→   436→      showCloseButton={false}
   437→   437→      className="max-w-xl top-[15%] translate-y-0"
   438→   438→    >
   439→   439→      <CommandInput
   440→   440→        value={query}
   441→   441→        onValueChange={setQuery}
   442→   442→        placeholder="Search projects, people, tags…"
   443→   443→        aria-label="Search query"
   444→   444→      />
   445→   445→      <CommandList>
   446→   446→        {q.length < 2 ? (
   447→   447→          <div className="py-8 px-4 text-center text-sm text-muted-foreground">
   448→   448→            Type at least 2 characters to search.
   449→   449→          </div>
   450→   450→        ) : searching ? (
   451→   451→          <div className="py-8 px-4 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   452→   452→            <Loader2 className="h-4 w-4 animate-spin" /> Searching…
   453→   453→          </div>
   454→   454→        ) : hasResults ? (
   455→   455→          <>
   456→   456→            {results!.projects.length > 0 && (
   457→   457→              <CommandGroup heading="Projects">
   458→   458→                {results!.projects.map((p) => (

   664→   664→export function AppShell({ children }: { children: React.ReactNode }) {
   665→   665→  const user = useApp((s) => s.user);
   666→   666→  const setUnreadCount = useApp((s) => s.setUnreadCount);
   667→   667→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   668→   668→  const userId = user?.id ?? null;
   669→   669→
   670→   670→  // Poll unread notification count every 45s while signed in.
   671→   671→  useEffect(() => {
   672→   672→    if (!userId) return;
   673→   673→    let cancelled = false;
   674→   674→    const poll = async () => {
   675→   675→      try {
   676→   676→        const res = await get<{ notifications: unknown[]; unread: number }>("/api/notifications");
   677→   677→        if (!cancelled) setUnreadCount(res.unread);
   678→   678→      } catch {
   679→   679→        /* silent — background badge poll */
   680→   680→      }
   681→   681→    };
   682→   682→    poll();
   683→   683→    const timer = setInterval(poll, 45_000);
   684→   684→    return () => {
   685→   685→      cancelled = true;
   686→   686→      clearInterval(timer);
   687→   687→    };
   688→   688→  }, [userId, setUnreadCount]);
   689→   689→
   690→   690→  // "/" opens global search (ignored while typing in form fields).
   691→   691→  useEffect(() => {
   692→   692→    const onKey = (e: KeyboardEvent) => {
   693→   693→      if (e.key !== "/") return;
   694→   694→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   695→   695→      const target = e.target as HTMLElement | null;
   696→   696→      const tag = target?.tagName;
   697→   697→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || target?.isContentEditable) return;
   698→   698→      e.preventDefault();
   699→   699→      setSearchOpen(true);
   700→   700→    };
   701→   701→    window.addEventListener("keydown", onKey);
   702→   702→    return () => window.removeEventListener("keydown", onKey);
   703→   703→  }, [setSearchOpen]);
   704→   704→
   705→   705→  return (
   706→   706→    <MotionConfig reducedMotion="user">
   707→   707→      <div className="min-h-screen flex">
   708→   708→        <LeftNav />
   709→   709→
   710→   710→        <div className="flex-1 flex flex-col min-w-0">
   711→   711→          <TopBar />
   712→   712→          <main className="flex-1 pb-24 md:pb-10" id="main-content">
   713→   713→            <div className="max-w-5xl mx-auto px-4 md:px-8 py-6">
   714→   714→              <motion.div
   715→   715→                initial={{ opacity: 0, y: 8 }}
   716→   716→                animate={{ opacity: 1, y: 0 }}
   717→   717→                transition={{ duration: 0.3, ease: "easeOut" }}
   718→   718→              >
   719→   719→                {children}
   720→   720→              </motion.div>
   721→   721→            </div>
   722→   722→          </main>
   723→   723→          <footer className="py-4 text-center text-xs text-muted-foreground/70 select-none" aria-label="Footer">
   724→   724→            TEKA NEXUS — demo preview
   725→   725→          </footer>
   726→   726→        </div>
   727→   727→
   728→   728→        <RightRail />
   729→   729→        <MobileNav />
   730→   730→        <SearchDialog />
   731→   731→      </div>
   732→   732→    </MotionConfig>
   733→   733→  );
