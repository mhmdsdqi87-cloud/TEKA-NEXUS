   183→function ProfileTab({ user }: { user: SessionUser }) {
   184→  const setUser = useApp((s) => s.setUser);
   185→  const t = useT();
   186→  const lbl = useLabeled();
   187→
   188→  const [displayName, setDisplayName] = useState(user.displayName);
   189→  const [bio, setBio] = useState(user.bio ?? "");
   190→  const [websiteUrl, setWebsiteUrl] = useState(user.websiteUrl ?? "");
   191→  const [avatarColor, setAvatarColor] = useState(user.avatarColor);
   192→  const [skills, setSkills] = useState<string[]>([]);
   193→  const [skillsLoaded, setSkillsLoaded] = useState(false);
   194→  const [saving, setSaving] = useState(false);
   195→  const [websiteError, setWebsiteError] = useState<string | null>(null);
   196→
   197→  // SessionUser doesn't carry skills — load them from the public profile.
   198→  useEffect(() => {
   199→    let alive = true;
   200→    get<{ user: ProfileDetail }>(`/api/users/${encodeURIComponent(user.username)}`)
   201→      .then((res) => {
   202→        if (!alive) return;
   203→        setSkills(res.user.skills);
   204→        setSkillsLoaded(true);
   205→      })
   206→      .catch(() => {
   207→        if (alive) setSkillsLoaded(true);
   208→      });
   209→    return () => {
   210→      alive = false;
   211→    };
   212→  }, [user.username]);
   213→
   214→  async function saveProfile() {
   215→    const name = displayName.trim();
   216→    if (!name) {
   217→      toast.error(t("settings.errors.emptyDisplayName"));
   218→      return;
   219→    }
   220→    const site = websiteUrl.trim();
   221→    if (site && !/^https?:\/\/\S+$/i.test(site)) {
   222→      setWebsiteError(t("settings.profileTab.websiteError"));
   223→      return;
   224→    }
   225→    setWebsiteError(null);
   226→    setSaving(true);
   227→    try {
   228→      const res = await patch<{ user: SessionUser }>("/api/me/profile", {
   229→        displayName: name,
   230→        bio,
   231→        websiteUrl: site,
   232→        avatarColor,
   233→        // Only replace skills once the current set has loaded, so we never wipe them.
   234→        ...(skillsLoaded ? { skills } : {}),
   235→      });
   236→      setUser(res.user);
   237→      toast.success(t("settings.toasts.profileUpdated"));
   238→    } catch (e) {
   239→      toast.error(e instanceof Error ? e.message : t("settings.errors.saveFailed"));
   240→    } finally {
   241→      setSaving(false);
   242→    }
   243→  }
   244→
   245→  function toggleSkill(slug: string) {
   246→    setSkills((prev) => (prev.includes(slug) ? prev.filter((s) => s !== slug) : [...prev, slug]));
   247→  }
   248→
   249→  return (
   250→    <div className="space-y-5">
   251→      <Reveal y={10}>
   252→        <section className="glass-card space-y-5 rounded-xl p-5">
   253→          <div className="space-y-2">
   254→            <Label htmlFor="display-name">{t("settings.profileTab.displayName")}</Label>
   255→          <Input
   256→            id="display-name"
   257→            value={displayName}
   258→            maxLength={60}
   259→            onChange={(e) => setDisplayName(e.target.value)}
   260→            className="focus-glow"
   261→            placeholder={t("settings.profileTab.displayNamePlaceholder")}
   262→          />
   263→        </div>
   264→
   265→        <div className="space-y-2">
   266→          <div className="flex items-center justify-between">
   267→            <Label htmlFor="bio">{t("settings.profileTab.bio")}</Label>
   268→            <span
   269→              className={cn(
   270→                "font-mono text-[10px]",
   271→                bio.length > LIMITS.bio ? "text-rose-300" : "text-muted-foreground"
   272→              )}
   273→            >
   274→              {bio.length} / {LIMITS.bio}
   275→            </span>
   276→          </div>
   277→          <Textarea
   278→            id="bio"
   279→            rows={4}
   280→            maxLength={LIMITS.bio}
   281→            value={bio}
   282→            onChange={(e) => setBio(e.target.value)}
   283→            placeholder={t("settings.profileTab.bioPlaceholder")}
   284→            className="focus-glow"
   285→          />
   286→        </div>
   287→
   288→        <div className="space-y-2">
   289→          <Label htmlFor="website">{t("settings.profileTab.website")}</Label>
   290→          <Input
   291→            id="website"
   292→            value={websiteUrl}
   293→            onChange={(e) => {
   294→              setWebsiteUrl(e.target.value);
   295→              setWebsiteError(null);
   296→            }}
   297→            placeholder="https://your-site.dev"
   298→            className={cn("font-mono text-sm focus-glow", websiteError && "border-rose-500/50")}
   299→          />
   300→          {websiteError ? (
   301→            <p className="text-xs text-rose-300">{websiteError}</p>
   302→          ) : (
   303→            <p className="text-xs text-muted-foreground">{t("settings.profileTab.websiteHint")}</p>
   304→          )}
   305→        </div>
   306→
   307→        <div className="space-y-2">
   308→          <Label>{t("settings.profileTab.avatarGradient")}</Label>
   309→          <div className="flex flex-wrap items-center gap-4">
   310→            <UserAvatar user={{ ...user, avatarColor }} size={56} />
   311→            <div className="flex flex-wrap gap-2" role="group" aria-label={t("settings.profileTab.chooseGradient")}>
   312→              {AVATAR_GRADIENT_KEYS.map((key) => (
   313→                <button
   314→                  key={key}
   315→                  type="button"
   316→                  onClick={() => setAvatarColor(key)}
   317→                  aria-label={t("settings.profileTab.gradientName", { name: key })}
   318→                  aria-pressed={avatarColor === key}
   319→                  className={cn(
   320→                    "h-9 w-9 rounded-full bg-gradient-to-br transition-all focus-glow",
   321→                    AVATAR_GRADIENTS[key],
   322→                    avatarColor === key
   323→                      ? "scale-110 ring-2 ring-cyan-300 ring-offset-2 ring-offset-[#0b0b12]"
   324→                      : "opacity-70 hover:scale-105 hover:opacity-100"
   325→                  )}
   326→                />
   327→              ))}
   328→            </div>
   329→          </div>
   330→        </div>
   331→      </section>
   332→      </Reveal>
   333→
   334→      <Reveal y={10} delay={0.06}>
   335→      <section className="glass-card space-y-3 rounded-xl p-5">
   336→        <div>
   337→          <Label>{t("settings.profileTab.skills")}</Label>
   338→          <p className="mt-1 text-xs text-muted-foreground">
   339→            {t("settings.profileTab.skillsHint")}
   340→          </p>
   341→        </div>
   342→        <div className="flex flex-wrap gap-2">
   343→          {SKILLS.map((s) => {
   344→            const selected = skills.includes(s.slug);
   345→            return (
   346→              <button
   347→                key={s.slug}
   348→                type="button"
   349→                onClick={() => toggleSkill(s.slug)}
   350→                aria-pressed={selected}
   351→                className={cn(
   352→                  "rounded-full border px-3 py-1.5 text-xs transition-colors focus-glow",
   353→                  selected
   354→                    ? "border-violet-500/40 bg-violet-500/15 text-violet-200"
   355→                    : "border-white/10 bg-white/[0.03] text-muted-foreground hover:border-white/20 hover:text-foreground"
   356→                )}
   357→              >
   358→                {lbl("project.skills", s.slug, s.name)}
   359→              </button>
   360→            );
   361→          })}
   362→        </div>
   363→      </section>
   364→      </Reveal>
   365→
   366→      <div className="flex justify-end">
   367→        <TapScale>
   368→          <Button onClick={saveProfile} disabled={saving} className="btn-hero min-w-32">
   369→            {saving && <Loader2 className="h-4 w-4 animate-spin" />}
   370→            {t("common.saveChanges")}
   371→          </Button>
   372→        </TapScale>
   373→      </div>
   374→    </div>
   375→  );
   376→}

   380→function AccountTab({ user }: { user: SessionUser }) {
   381→  const setUser = useApp((s) => s.setUser);
   382→  const navigate = useApp((s) => s.navigate);
   383→  const t = useT();
   384→  const locale = useI18n((s) => s.locale);
   385→  const activeLocaleNative = LOCALES.find((l) => l.code === locale)?.native ?? "English";
   386→
   387→  const [pwCurrent, setPwCurrent] = useState("");
   388→  const [pwNew, setPwNew] = useState("");
   389→  const [pwConfirm, setPwConfirm] = useState("");
   390→  const [pwPending, setPwPending] = useState(false);
   391→
   392→  const [delOpen, setDelOpen] = useState(false);
   393→  const [delStage, setDelStage] = useState<"username" | "password">("username");
   394→  const [delConfirm, setDelConfirm] = useState("");
   395→  const [delPassword, setDelPassword] = useState("");
   396→  const [delPending, setDelPending] = useState(false);
   397→  const [delError, setDelError] = useState<string | null>(null);
   398→
   399→  const pwTooShort = pwNew.length > 0 && pwNew.length < 8;
   400→  const pwMismatch = pwConfirm.length > 0 && pwNew !== pwConfirm;
   401→
   402→  async function changePassword() {
   403→    if (!pwCurrent) {
   404→      toast.error(t("settings.errors.enterCurrentPassword"));
   405→      return;
   406→    }
   407→    if (pwNew.length < 8) {
   408→      toast.error(t("settings.errors.pwTooShort"));
   409→      return;
   410→    }
   411→    if (pwNew !== pwConfirm) {
   412→      toast.error(t("settings.errors.pwMismatch"));
   413→      return;
   414→    }
   415→    setPwPending(true);
   416→    try {
   417→      await patch("/api/me/password", { currentPassword: pwCurrent, newPassword: pwNew });
   418→      toast.success(t("settings.toasts.passwordChanged"));
   419→      setPwCurrent("");
   420→      setPwNew("");
   421→      setPwConfirm("");
   422→    } catch (e) {
   423→      toast.error(e instanceof Error ? e.message : t("settings.errors.pwChangeFailed"));
   424→    } finally {
   425→      setPwPending(false);
   426→    }
   427→  }
   428→
   429→  async function deleteAccount() {
   430→    setDelPending(true);
   431→    setDelError(null);
   432→    try {
   433→      await del("/api/me/account", { password: delPassword });
   434→      setDelOpen(false);
   435→      setUser(null);
   436→      toast.success(t("settings.toasts.accountDeleted"));
   437→      navigate({ name: "landing" });
   438→    } catch (e) {
   439→      setDelError(e instanceof Error ? e.message : t("settings.errors.deleteFailed"));
   440→    } finally {
   441→      setDelPending(false);
   442→    }
   443→  }
   444→
   445→  function closeDeleteDialog(open: boolean) {
   446→    setDelOpen(open);
   447→    if (!open) {
   448→      setDelStage("username");
   449→      setDelConfirm("");
   450→      setDelPassword("");
   451→      setDelError(null);
   452→    }
   453→  }
   454→
   455→  return (
   456→    <div className="space-y-5">
   457→      <Reveal y={10}>
   458→      <section className="glass-card space-y-4 rounded-xl p-5">
   459→        <div className="flex items-center justify-between gap-4 flex-wrap">
   460→          <div>
   461→            <h3 className="font-display font-semibold">{t("settings.accountTab.language")}</h3>
   462→            <p className="mt-0.5 text-xs text-muted-foreground">
   463→              {t("settings.accountTab.languageDescription")}
   464→            </p>
   465→          </div>
   466→          <div className="flex items-center gap-3">
   467→            <span className="hidden sm:block text-sm text-muted-foreground" aria-hidden="true">
   468→              {activeLocaleNative}
   469→            </span>
   470→            <LanguageSwitcher />
   471→          </div>
   472→        </div>
   473→      </section>
   474→      </Reveal>
   475→
   476→      <Reveal y={10} delay={0.05}>
   477→      <section className="glass-card space-y-5 rounded-xl p-5">
   478→        <div className="space-y-2">
   479→          <Label htmlFor="account-email">{t("settings.accountTab.email")}</Label>
   480→          <Input
   481→            id="account-email"
   482→            value={user.email}
   483→            readOnly
   484→            className="cursor-default font-mono text-sm focus-glow"
   485→            aria-describedby="account-email-hint"
   486→          />
   487→          <p id="account-email-hint" className="text-xs text-muted-foreground">
   488→            {t("settings.accountTab.emailHint")}
   489→          </p>
   490→        </div>
   491→      </section>
   492→      </Reveal>
   493→
   494→      <Reveal y={10} delay={0.1}>
   495→      <section className="glass-card space-y-4 rounded-xl p-5">
   496→        <div>
   497→          <h3 className="font-display font-semibold">{t("settings.accountTab.changePassword")}</h3>
   498→          <p className="mt-0.5 text-xs text-muted-foreground">
   499→            {t("settings.accountTab.changePasswordHint")}
   500→          </p>
   501→        </div>
   502→        <div className="grid gap-4 sm:grid-cols-3">
   503→          <div className="space-y-2">
   504→            <Label htmlFor="pw-current">{t("settings.accountTab.currentPassword")}</Label>
   505→            <Input
   506→              id="pw-current"
   507→              type="password"
   508→              autoComplete="current-password"
   509→              value={pwCurrent}
   510→              onChange={(e) => setPwCurrent(e.target.value)}
   511→              className="focus-glow"
   512→            />
   513→          </div>
   514→          <div className="space-y-2">
   515→            <Label htmlFor="pw-new">{t("settings.accountTab.newPassword")}</Label>
   516→            <Input
   517→              id="pw-new"
   518→              type="password"
   519→              autoComplete="new-password"
   520→              value={pwNew}
   521→              onChange={(e) => setPwNew(e.target.value)}
   522→              className={cn("focus-glow", pwTooShort && "border-rose-500/50")}
   523→            />
   524→            {pwTooShort && <p className="text-xs text-rose-300">{t("settings.accountTab.pwTooShort")}</p>}
   525→          </div>
   526→          <div className="space-y-2">
   527→            <Label htmlFor="pw-confirm">{t("settings.accountTab.confirmPassword")}</Label>
   528→            <Input
   529→              id="pw-confirm"
   530→              type="password"
   531→              autoComplete="new-password"
   532→              value={pwConfirm}
   533→              onChange={(e) => setPwConfirm(e.target.value)}
   534→              className={cn("focus-glow", pwMismatch && "border-rose-500/50")}
   535→            />
   536→            {pwMismatch && <p className="text-xs text-rose-300">{t("settings.accountTab.pwMismatch")}</p>}
   537→          </div>
   538→        </div>
   539→        <div className="flex justify-end">
   540→          <TapScale>
   541→            <Button
   542→              onClick={changePassword}
   543→              disabled={pwPending || !pwCurrent || !pwNew || !pwConfirm}
   544→              className="min-w-36"
   545→            >
   546→              {pwPending && <Loader2 className="h-4 w-4 animate-spin" />}
   547→              {t("settings.accountTab.updatePassword")}
   548→            </Button>
   549→          </TapScale>
   550→        </div>
   551→      </section>
   552→      </Reveal>
   553→
   554→      {/* Danger zone */}
   555→      <section className="rounded-xl border border-rose-500/25 bg-rose-500/[0.04] p-5">
   556→        <h3 className="font-display font-semibold text-rose-200">{t("settings.accountTab.dangerZone")}</h3>
   557→        <p className="mt-1 max-w-lg text-sm text-muted-foreground">
   558→          {t("settings.accountTab.dangerDescription")}
   559→        </p>
   560→        <AlertDialog open={delOpen} onOpenChange={closeDeleteDialog}>
   561→          <AlertDialogTrigger asChild>
   562→            <Button variant="destructive" className="mt-4">
   563→              <Trash2 className="h-4 w-4" />
   564→              {t("settings.accountTab.deleteAccount")}
   565→            </Button>
   566→          </AlertDialogTrigger>
   567→          <AlertDialogContent>
   568→            {delStage === "username" ? (
   569→              <>
   570→                <AlertDialogHeader>
   571→                  <AlertDialogTitle className="font-display">{t("settings.delete.title")}</AlertDialogTitle>
   572→                  <AlertDialogDescription>
   573→                    {t("settings.delete.typeUsername")}{" "}
   574→                    <span className="font-mono text-foreground">{user.username}</span>{" "}
   575→                    {t("settings.delete.toContinue")}
   576→                  </AlertDialogDescription>
   577→                </AlertDialogHeader>
   578→                <Input
   579→                  value={delConfirm}
   580→                  onChange={(e) => setDelConfirm(e.target.value)}
   581→                  placeholder={user.username}
   582→                  autoComplete="off"
   583→                  className="font-mono focus-glow"
   584→                  aria-label={t("settings.delete.usernameAria")}
   585→                />
   586→                <AlertDialogFooter>
   587→                  <AlertDialogCancel>{t("common.cancel")}</AlertDialogCancel>
   588→                  <AlertDialogAction
   589→                    disabled={delConfirm !== user.username}
   590→                    className="bg-rose-600 text-white hover:bg-rose-600/90"
   591→                    onClick={(e) => {
   592→                      e.preventDefault(); // keep the dialog open, advance to password step
   593→                      setDelStage("password");
   594→                    }}
   595→                  >
   596→                    {t("settings.delete.continue")}
   597→                  </AlertDialogAction>
   598→                </AlertDialogFooter>
   599→              </>
   600→            ) : (
   601→              <>
   602→                <AlertDialogHeader>
   603→                  <AlertDialogTitle className="font-display">
   604→                    {t("settings.delete.confirmTitle")}
   605→                  </AlertDialogTitle>
   606→                  <AlertDialogDescription>
   607→                    {t("settings.delete.confirmDescription")}
   608→                  </AlertDialogDescription>
   609→                </AlertDialogHeader>
   610→                <Input
   611→                  type="password"
   612→                  autoComplete="current-password"
   613→                  value={delPassword}
   614→                  onChange={(e) => {
   615→                    setDelPassword(e.target.value);
   616→                    setDelError(null);
   617→                  }}
   618→                  onKeyDown={(e) => {
   619→                    if (e.key === "Enter" && delPassword && !delPending) deleteAccount();
   620→                  }}
   621→                  placeholder={t("settings.delete.passwordPlaceholder")}
   622→                  className="focus-glow"
   623→                  aria-label={t("settings.delete.passwordAria")}
   624→                />
   625→                {delError && <p className="text-sm text-rose-300">{delError}</p>}
   626→                <AlertDialogFooter>
   627→                  <AlertDialogCancel onClick={() => setDelStage("username")}>
   628→                    {t("common.back")}
   629→                  </AlertDialogCancel>
   630→                  <AlertDialogAction
   631→                    disabled={delPending || !delPassword}
   632→                    className="bg-rose-600 text-white hover:bg-rose-600/90"
   633→                    onClick={(e) => {
   634→                      e.preventDefault(); // keep dialog open until the request succeeds
   635→                      deleteAccount();
   636→                    }}
   637→                  >
   638→                    {delPending ? (
   639→                      <Loader2 className="h-4 w-4 animate-spin" />
   640→                    ) : (
   641→                      <Trash2 className="h-4 w-4" />
   642→                    )}
   643→                    {t("settings.delete.deleteForever")}
   644→                  </AlertDialogAction>
   645→                </AlertDialogFooter>
   646→              </>
   647→            )}
   648→          </AlertDialogContent>
   649→        </AlertDialog>
   650→      </section>
   651→    </div>
   652→  );

   655→// ─── NOTIFICATIONS ──────────────────────────────────────────────
   656→
   657→function NotificationsTab() {
   658→  const t = useT();
   659→  const [prefs, setPrefs] = useState<Prefs | null>(null);
   660→  const [pendingField, setPendingField] = useState<string | null>(null);
   661→
   662→  useEffect(() => {
   663→    let alive = true;
   664→    get<{ prefs: Prefs }>("/api/me/settings")
   665→      .then((res) => {
   666→        if (alive) setPrefs(res.prefs);
   667→      })
   668→      .catch((e) => {
   669→        toast.error(e instanceof Error ? e.message : t("settings.errors.loadPrefsFailed"));
   670→      });
   671→    return () => {
   672→      alive = false;
   673→    };
   674→  }, []);
   675→
   676→  async function togglePref(field: keyof Prefs, value: boolean) {
   677→    if (!prefs) return;
   678→    const previous = prefs;
   679→    setPrefs({ ...prefs, [field]: value }); // optimistic
   680→    setPendingField(field);
   681→    try {
   682→      const res = await patch<{ prefs: Prefs }>("/api/me/settings", { [field]: value });
   683→      setPrefs(res.prefs);
   684→      toast.success(t("settings.toasts.prefsUpdated"));
   685→    } catch (e) {
   686→      setPrefs(previous); // revert on error
   687→      toast.error(e instanceof Error ? e.message : t("settings.errors.prefsUpdateFailed"));
   688→    } finally {
   689→      setPendingField(null);
   690→    }
   691→  }
   692→
   693→  if (!prefs) {
   694→    return (
   695→      <div className="space-y-3">
   696→        {Array.from({ length: 3 }).map((_, i) => (
   697→          <Skeleton key={i} className="h-[76px] w-full rounded-xl bg-white/[0.05]" />
   698→        ))}
   699→      </div>
   700→    );
   701→  }
   702→
   703→  const rows: { field: keyof Prefs; title: string; description: string }[] = [
   704→    {
   705→      field: "projectActivity",
   706→      title: t("settings.notificationsTab.projectActivity.title"),
   707→      description: t("settings.notificationsTab.projectActivity.description"),
   708→    },
   709→    {
   710→      field: "collaborationRequests",
   711→      title: t("settings.notificationsTab.collaboration.title"),
   712→      description: t("settings.notificationsTab.collaboration.description"),
   713→    },
   714→    {
   715→      field: "socialActivity",
   716→      title: t("settings.notificationsTab.social.title"),
   717→      description: t("settings.notificationsTab.social.description"),
   718→    },
   719→  ];
   720→
   721→  return (
   722→    <div className="space-y-3">
   723→      {rows.map((row, i) => (
   724→        <Reveal key={row.field} y={10} delay={i * 0.05}>
   725→          <div
   726→            className="glass-card flex items-center justify-between gap-4 rounded-xl p-4"
   727→          >
   728→            <div className="min-w-0">
   729→              <p className="text-sm font-medium">{row.title}</p>
   730→              <p className="mt-0.5 text-xs leading-relaxed text-muted-foreground">
   731→                {row.description}
   732→              </p>
   733→            </div>
   734→            <Switch
   735→              checked={prefs[row.field]}
   736→              disabled={pendingField !== null}
   737→              onCheckedChange={(v) => togglePref(row.field, v)}
   738→              aria-label={t("settings.notificationsTab.toggleAria", { name: row.title })}
   739→            />
   740→          </div>
   741→        </Reveal>
   742→      ))}
   743→    </div>
   744→  );
   745→}
