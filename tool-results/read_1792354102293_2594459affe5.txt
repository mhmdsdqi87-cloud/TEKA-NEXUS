   384→export function SearchDialog() {
   385→  const open = useApp((s) => s.searchOpen);
   386→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   387→  const navigate = useApp((s) => s.navigate);
   388→
   389→  const [query, setQuery] = useState("");
   390→  const [data, setData] = useState<{ q: string; results: SearchResults } | null>(null);
   391→  const reqId = useRef(0);
   392→
   393→  // Debounced server search (250ms, min 2 chars).
   394→  useEffect(() => {
   395→    const q = query.trim();
   396→    if (q.length < 2) return;
   397→    const id = ++reqId.current;
   398→    const timer = setTimeout(async () => {
   399→      try {
   400→        const res = await get<SearchResults>(`/api/search${qs({ q })}`);
   401→        if (reqId.current === id) setData({ q, results: res });
   402→      } catch {
   403→        if (reqId.current === id) setData({ q, results: { projects: [], users: [], tags: [] } });
   404→      }
   405→    }, 250);
   406→    return () => clearTimeout(timer);
   407→  }, [query]);
   408→
   409→  // Close + reset (event handler, not an effect).
   410→  const handleOpenChange = (next: boolean) => {
   411→    setSearchOpen(next);
   412→    if (!next) {
   413→      reqId.current++;
   414→      setQuery("");
   415→      setData(null);
   416→    }
   417→  };
   418→
   419→  const q = query.trim();
   420→  // Results are only shown when they belong to the current query;
   421→  // a newer query with no response yet renders the "Searching…" state.
   422→  const results = data && data.q === q ? data.results : null;
   423→  const searching = q.length >= 2 && results === null;
   424→  const hasResults =
   425→    results && (results.projects.length > 0 || results.users.length > 0 || results.tags.length > 0);
   426→
   427→  return (
   428→    <CommandDialog
   429→      open={open}
   430→      onOpenChange={handleOpenChange}
   431→      title="Search TEKA NEXUS"
   432→      description="Search projects, people and tags"
   433→      showCloseButton={false}
   434→      className="max-w-xl top-[15%] translate-y-0"
   435→    >
   436→      <CommandInput
   437→        value={query}
   438→        onValueChange={setQuery}
   439→        placeholder="Search projects, people, tags…"
   440→        aria-label="Search query"
   441→      />
   442→      <CommandList>
   443→        {q.length < 2 ? (
   444→          <div className="py-8 px-4 text-center text-sm text-muted-foreground">
   445→            Type at least 2 characters to search.
   446→          </div>
   447→        ) : searching ? (
   448→          <div className="py-8 px-4 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   449→            <Loader2 className="h-4 w-4 animate-spin" /> Searching…
   450→          </div>
   451→        ) : hasResults ? (
   452→          <>
   453→            {results!.projects.length > 0 && (
   454→              <CommandGroup heading="Projects">
   455→                {results!.projects.map((p) => (
   456→                  <CommandItem
   457→                    key={p.id}
   458→                    value={`${q} ${p.name}`}
   459→                    onSelect={() => {
   460→                      setSearchOpen(false);
   461→                      navigate({ name: "project", slug: p.slug });
   462→                    }}
   463→                  >
   464→                    <FileCode2 className="text-cyan-300" />
   465→                    <span className="truncate">{p.name}</span>
   466→                    <span className="ml-auto truncate max-w-[45%] text-xs text-muted-foreground">{p.tagline}</span>
   467→                  </CommandItem>
   468→                ))}
   469→              </CommandGroup>
   470→            )}
   471→            {results!.users.length > 0 && (
   472→              <CommandGroup heading="Users">
   473→                {results!.users.map((u) => (
   474→                  <CommandItem
   475→                    key={u.id}
   476→                    value={`${q} ${u.displayName} ${u.username}`}
   477→                    onSelect={() => {
   478→                      setSearchOpen(false);
   479→                      navigate({ name: "profile", username: u.username });
   480→                    }}
   481→                  >
   482→                    <UserAvatar user={u} size={20} />
   483→                    <span className="truncate">{u.displayName}</span>
   484→                    <span className="ml-auto text-xs text-muted-foreground">@{u.username}</span>
   485→                  </CommandItem>
   486→                ))}
   487→              </CommandGroup>
   488→            )}
   489→            {results!.tags.length > 0 && (
   490→              <CommandGroup heading="Tags">
   491→                {results!.tags.map((t) => (
   492→                  <CommandItem
   493→                    key={t.tag}
   494→                    value={`${q} ${t.tag}`}
   495→                    onSelect={() => {
   496→                      setSearchOpen(false);
   497→                      navigate({ name: "explore", query: t.tag });
   498→                    }}
   499→                  >
   500→                    <Hash className="text-violet-300" />
   501→                    <span className="font-mono">#{t.tag}</span>
   502→                    <span className="ml-auto text-xs text-muted-foreground">{t.count} project{t.count === 1 ? "" : "s"}</span>
   503→                  </CommandItem>
   504→                ))}
   505→              </CommandGroup>
   506→            )}
   507→          </>
   508→        ) : (
   509→          <CommandEmpty>No results for &ldquo;{q}&rdquo;</CommandEmpty>
   510→        )}
   511→      </CommandList>
   512→    </CommandDialog>
   513→  );
   514→}

   526→function RightRail() {

   528→  const navigate = useApp((s) => s.navigate);
   529→
   530→  const [trending, setTrending] = useState<ProjectCard[] | null>(null);
   531→  const [requests, setRequests] = useState<{ ownerId: string; received: number; sent: number } | null>(
   532→    null
   533→  );
   534→
   535→  const userId = user?.id ?? null;
   536→
   537→  // Trending projects (public, popular first, top 3).
   538→  useEffect(() => {
   539→    let cancelled = false;
   540→    get<{ projects: ProjectCard[]; page: number; pages: number; total: number }>(
   541→      `/api/projects${qs({ sort: "popular", page: 1 })}`
   542→    )
   543→      .then((res) => {
   544→        if (!cancelled) setTrending(res.projects.slice(0, 3));
   545→      })
   546→      .catch((e) => {
   547→        if (!cancelled) toast.error(e instanceof Error ? e.message : "Failed to load trending projects");
   548→      });
   549→    return () => {
   550→      cancelled = true;
   551→    };
   552→  }, []);
   553→
   554→  // Pending team invitations & requests (authed only).

   661→export function AppShell({ children }: { children: React.ReactNode }) {
   662→  const user = useApp((s) => s.user);
   663→  const setUnreadCount = useApp((s) => s.setUnreadCount);
   664→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   665→  const userId = user?.id ?? null;
   666→
   667→  // Poll unread notification count every 45s while signed in.
   668→  useEffect(() => {
   669→    if (!userId) return;
   670→    let cancelled = false;
   671→    const poll = async () => {
   672→      try {
   673→        const res = await get<{ notifications: unknown[]; unread: number }>("/api/notifications");
   674→        if (!cancelled) setUnreadCount(res.unread);
   675→      } catch {
   676→        /* silent — background badge poll */
   677→      }
   678→    };
   679→    poll();
   680→    const timer = setInterval(poll, 45_000);
   681→    return () => {
   682→      cancelled = true;
   683→      clearInterval(timer);
   684→    };
   685→  }, [userId, setUnreadCount]);
   686→
   687→  // "/" opens global search (ignored while typing in form fields).
   688→  useEffect(() => {
   689→    const onKey = (e: KeyboardEvent) => {
   690→      if (e.key !== "/") return;
   691→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   692→      const target = e.target as HTMLElement | null;
   693→      const tag = target?.tagName;
   694→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || target?.isContentEditable) return;
   695→      e.preventDefault();
   696→      setSearchOpen(true);
   697→    };
   698→    window.addEventListener("keydown", onKey);
   699→    return () => window.removeEventListener("keydown", onKey);
   700→  }, [setSearchOpen]);
   701→
   702→  return (
   703→    <MotionConfig reducedMotion="user">
   704→      <div className="min-h-screen flex">
   705→        <LeftNav />
   706→
   707→        <div className="flex-1 flex flex-col min-w-0">
   708→          <TopBar />
   709→          <main className="flex-1 pb-24 md:pb-10" id="main-content">
   710→            <div className="max-w-5xl mx-auto px-4 md:px-8 py-6">
   711→              <motion.div
   712→                initial={{ opacity: 0, y: 8 }}
   713→                animate={{ opacity: 1, y: 0 }}
   714→                transition={{ duration: 0.3, ease: "easeOut" }}
   715→              >
   716→                {children}
   717→              </motion.div>
   718→            </div>
   719→          </main>
   720→          <footer className="py-4 text-center text-xs text-muted-foreground/70 select-none" aria-label="Footer">
   721→            TEKA NEXUS — demo preview
   722→          </footer>
   723→        </div>
   724→
   725→        <RightRail />
   726→        <MobileNav />
   727→        <SearchDialog />
   728→      </div>
   729→    </MotionConfig>
   730→  );
   731→}
