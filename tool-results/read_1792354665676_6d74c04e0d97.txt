    69→  projects: ProjectCard[];

    73→const CONTRIBUTION_META: Record<string, { icon: LucideIcon; label: string }> = {
    74→  TASK_COMPLETED: { icon: CheckCircle2, label: "Completed task" },
    75→  RELEASE_PUBLISHED: { icon: Rocket, label: "Published release" },
    76→  MILESTONE_COMPLETED: { icon: Flag, label: "Completed milestone" },
    77→  UPDATE_POSTED: { icon: FileText, label: "Posted update" },

    79→  PROJECT_CREATED: { icon: Sparkles, label: "Created project" },
    80→};

    82→const REPORT_REASONS = [
    83→  { value: "SPAM", label: "Spam" },
    84→  { value: "HARASSMENT", label: "Harassment" },
    85→  { value: "INAPPROPRIATE", label: "Inappropriate content" },
    86→  { value: "COPYRIGHT", label: "Copyright violation" },
    87→  { value: "OTHER", label: "Something else" },

    90→function ChipRow({
    91→  label,
    92→  items,
    93→  tone,
    94→}: {
    95→  label: string;
    96→  items: string[];
    97→  tone: "default" | "cyan" | "violet";
    98→}) {
    99→  if (items.length === 0) return null;
   100→  const toneClass =
   101→    tone === "violet"
   102→      ? "border-violet-500/25 bg-violet-500/10 text-violet-300"
   103→      : tone === "cyan"
   104→        ? "border-cyan-500/20 bg-cyan-500/10 text-cyan-200"
   105→        : "border-white/10 bg-white/[0.04] text-foreground/80";
   106→  return (
   107→    <div className="flex flex-wrap items-center gap-2">
   108→      <span className="font-mono text-[10px] uppercase tracking-widest text-muted-foreground/80">
   109→        {label}
   110→      </span>
   111→      {items.map((item) => (
   112→        <span key={item} className={cn("rounded-full border px-2.5 py-0.5 text-xs", toneClass)}>
   113→          {item}
   114→        </span>
   115→      ))}
   116→    </div>
   117→  );
   118→}
   119→
   120→function StatCard({ label, value }: { label: string; value: number }) {
   121→  return (
   122→    <div className="glass-card rounded-xl p-4">
   123→      <p className="font-display text-2xl font-bold tabular-nums">{value.toLocaleString()}</p>
   124→      <p className="mt-0.5 text-xs text-muted-foreground">{label}</p>
   125→    </div>
   126→  );
   127→}
   128→
   129→export function ProfileView({ username }: { username: string }) {
   130→  const user = useApp((s) => s.user);
   131→  const navigate = useApp((s) => s.navigate);
   132→
   133→  const [data, setData] = useState<ProfileResponse | null>(null);
   134→  const [loading, setLoading] = useState(true);
   135→  const [notFound, setNotFound] = useState(false);
   136→  const [loadFailed, setLoadFailed] = useState(false);
   137→
   138→  const [followPending, setFollowPending] = useState(false);
   139→  const [blockedByMe, setBlockedByMe] = useState(false);
   140→  const [blockPending, setBlockPending] = useState(false);
   141→  const [menuOpen, setMenuOpen] = useState(false);
   142→
   143→  const [reportOpen, setReportOpen] = useState(false);
   144→  const [reportReason, setReportReason] = useState<string>("SPAM");
   145→  const [reportDetails, setReportDetails] = useState("");
   146→  const [reportPending, setReportPending] = useState(false);
   147→
   148→  const load = useCallback(
   149→    async (silent = false) => {
   150→      if (!silent) {
   151→        setLoading(true);
   152→        setData(null);
   153→      }
   154→      setNotFound(false);
   155→      setLoadFailed(false);
   156→      try {
   157→        const res = await get<ProfileResponse>(`/api/users/${encodeURIComponent(username)}`);
   158→        setData(res);
   159→      } catch (e) {
   160→        if (e instanceof ApiError && e.status === 404) setNotFound(true);
   161→        else {
   162→          setLoadFailed(true);
   163→          toast.error(e instanceof Error ? e.message : "Couldn't load this profile.");
   164→        }
   165→      } finally {
   166→        setLoading(false);
   167→      }
   168→    },
   169→    [username]
   170→  );
   171→
   172→  useEffect(() => {
   173→    load();
   174→  }, [load]);
   175→
   176→  const profile = data?.user ?? null;
   177→
   178→  async function toggleFollow() {
   179→    if (!profile || profile.isSelf) return;
   180→    if (!user) {
   181→      toast.error("Sign in to follow creators.");
   182→      navigate({ name: "login" });
   183→      return;
   184→    }
   185→    setFollowPending(true);
   186→    try {
   187→      const res = await post<{ following: boolean; followerCount: number }>(
   188→        `/api/users/${encodeURIComponent(profile.username)}/follow`
   189→      );
   190→      setData((prev) =>
   191→        prev
   192→          ? {
   193→              ...prev,
   194→              user: { ...prev.user, isFollowing: res.following, followerCount: res.followerCount },
   195→            }
   196→          : prev
   197→      );
   198→      toast.success(res.following ? `Following ${profile.displayName}` : `Unfollowed ${profile.displayName}`);
   199→    } catch (e) {
   200→      toast.error(e instanceof Error ? e.message : "Couldn't update follow.");
   201→    } finally {
   202→      setFollowPending(false);
   203→    }
   204→  }
   205→
   206→  async function toggleBlock() {
   207→    if (!profile) return;
   208→    if (!user) {
   209→      toast.error("Sign in to manage blocks.");
   210→      navigate({ name: "login" });
   211→      return;
   212→    }
   213→    setBlockPending(true);
   214→    try {
   215→      const res = await post<{ blocked: boolean }>(
   216→        `/api/users/${encodeURIComponent(profile.username)}/block`
   217→      );
   218→      setBlockedByMe(res.blocked);
   219→      setMenuOpen(false);
   220→      toast.success(res.blocked ? `Blocked @${profile.username}` : `Unblocked @${profile.username}`);
   221→      // Refresh silently — blocking removes follows in both directions.
   222→      await load(true);
   223→    } catch (e) {
   224→      toast.error(e instanceof Error ? e.message : "Couldn't update block.");
   225→    } finally {
   226→      setBlockPending(false);
   227→    }
   228→  }
   229→
   230→  async function submitReport() {
   231→    if (!profile) return;
   232→    setReportPending(true);
   233→    try {
   234→      await post("/api/reports", {
   235→        targetType: "user",
   236→        targetId: profile.id,
   237→        reason: reportReason,
   238→        ...(reportDetails.trim() ? { details: reportDetails.trim() } : {}),
   239→      });
   240→      setReportOpen(false);
   241→      setReportDetails("");
   242→      setReportReason("SPAM");
   243→      toast.success("Report submitted — our team will review it.");
   244→    } catch (e) {
   245→      toast.error(e instanceof Error ? e.message : "Couldn't submit report.");
   246→    } finally {
   247→      setReportPending(false);
   248→    }
   249→  }
   250→
   251→  if (loading) {
   252→    return (
   253→      <div className="mx-auto w-full max-w-5xl space-y-6 px-4 py-6 md:py-8">
   254→        <Skeleton className="h-48 w-full rounded-2xl bg-white/[0.05]" />
   255→        <div className="grid grid-cols-2 gap-4 md:grid-cols-4">
   256→          {Array.from({ length: 4 }).map((_, i) => (
   257→            <Skeleton key={i} className="h-20 rounded-xl bg-white/[0.05]" />
   258→          ))}
   259→        </div>
   260→        <Skeleton className="h-44 w-full rounded-2xl bg-white/[0.05]" />
   261→      </div>
   262→    );
   263→  }
   264→
   265→  if (notFound) {
   266→    return (
   267→      <div className="mx-auto w-full max-w-5xl px-4 py-6 md:py-8">
   268→        <EmptyState
   269→          icon={UserX}
   270→          title="Creator not found"
   271→          description="This profile doesn't exist or may have been removed."
   272→          action={
   273→            <Button variant="outline" onClick={() => navigate({ name: "explore" })}>
   274→              Explore projects
   275→            </Button>
   276→          }
   277→        />
   278→      </div>
   279→    );
   280→  }
   281→
   282→  if (loadFailed || !profile) {
   283→    return (
   284→      <div className="mx-auto w-full max-w-5xl px-4 py-6 md:py-8">
   285→        <EmptyState
   286→          icon={Info}
   287→          title="Couldn't load this profile"
   288→          description="Check your connection and try again."
   289→          action={
   290→            <Button variant="outline" onClick={() => load()}>
   291→              Try again
   292→            </Button>
   293→          }
   294→        />
   295→      </div>
   296→    );
   297→  }
   298→
   299→  const goalNames = profile.goals.map((g) => GOALS.find((x) => x.slug === g)?.name ?? g);
   300→  const interestNames = profile.interests.map(
   301→    (i) => PROJECT_CATEGORIES.find((c) => c.slug === i)?.name ?? i
   302→  );
   303→  const skillNames = profile.skills.map((s) => SKILLS.find((x) => x.slug === s)?.name ?? s);
   304→  const websiteHost = profile.websiteUrl
   305→    ? profile.websiteUrl.replace(/^https?:\/\//i, "").replace(/\/$/, "")
   306→    : "";
   307→
   308→  return (
   309→    <MotionConfig reducedMotion="user">
   310→      <div className="mx-auto w-full max-w-5xl space-y-6 px-4 py-6 md:py-8">
   311→        {/* ── Header ─────────────────────────────────────────── */}
   312→        <motion.section
   313→          initial={{ opacity: 0, y: 10 }}
   314→          animate={{ opacity: 1, y: 0 }}
   315→          transition={{ duration: 0.3, ease: "easeOut" }}
   316→          className="aurora-border rounded-2xl p-5 md:p-7"
   317→        >
   318→          <div className="flex flex-col gap-6 md:flex-row md:items-start md:justify-between">
   319→            <div className="flex min-w-0 flex-1 flex-col gap-5 sm:flex-row sm:items-start">
   320→              <UserAvatar user={profile} size={72} />
   321→              <div className="min-w-0 flex-1 space-y-3">
   322→                <div>
   323→                  <h1 className="font-display text-2xl font-bold leading-tight md:text-3xl">
   324→                    {profile.displayName}
   325→                  </h1>
   326→                  <p className="font-mono text-sm text-muted-foreground">@{profile.username}</p>
   327→                </div>
   328→
   329→                {profile.bio && (
   330→                  <p className="max-w-xl whitespace-pre-line text-sm leading-relaxed text-foreground/85">
   331→                    {profile.bio}
   332→                  </p>
   333→                )}
   334→
   335→                {websiteHost && (
   336→                  <a
   337→                    href={profile.websiteUrl}
   338→                    target="_blank"
   339→                    rel="noopener noreferrer"
   340→                    className="inline-flex items-center gap-1.5 rounded text-sm text-cyan-300 transition-colors hover:text-cyan-200 hover:underline focus-glow"
   341→                  >
   342→                    <ExternalLink className="h-3.5 w-3.5" />
   343→                    <span className="max-w-[240px] truncate">{websiteHost}</span>
   344→                  </a>
   345→                )}
   346→
   347→                <div className="flex flex-wrap items-center gap-x-4 gap-y-2 text-xs text-muted-foreground">
   348→                  <ReputationChip xp={profile.xp} reputation={profile.reputation} />
   349→                  <span className="inline-flex items-center gap-1.5">
   350→                    <CalendarDays className="h-3.5 w-3.5" />
   351→                    Member since {format(new Date(profile.createdAt), "MMM yyyy")}
   352→                  </span>
   353→                </div>
   354→
   355→                <div className="space-y-2 pt-1">
   356→                  <ChipRow label="Goals" items={goalNames} tone="default" />
   357→                  <ChipRow label="Interests" items={interestNames} tone="cyan" />
   358→                  <ChipRow label="Skills" items={skillNames} tone="violet" />
   359→                </div>
   360→              </div>
   361→            </div>
   362→
   363→            <div className="flex shrink-0 items-center gap-2">
   364→              {profile.isSelf ? (
   365→                <Button onClick={() => navigate({ name: "settings", tab: "profile" })} className="btn-hero">
   366→                  <Settings2 className="h-4 w-4" />
   367→                  Edit profile
   368→                </Button>
   369→              ) : (
   370→                <>
   371→                  <Button
   372→                    onClick={toggleFollow}
   373→                    disabled={followPending || blockedByMe}
   374→                    variant={profile.isFollowing ? "outline" : "default"}
   375→                    className={cn("min-w-32", !profile.isFollowing && "btn-hero")}
   376→                  >
   377→                    {followPending ? (
   378→                      <Loader2 className="h-4 w-4 animate-spin" />
   379→                    ) : profile.isFollowing ? (
   380→                      <UserCheck className="h-4 w-4" />
   381→                    ) : (
   382→                      <UserPlus className="h-4 w-4" />
   383→                    )}
   384→                    {blockedByMe ? "Blocked" : profile.isFollowing ? "Following" : "Follow"}
   385→                  </Button>
   386→                  <DropdownMenu open={menuOpen} onOpenChange={setMenuOpen}>
   387→                    <DropdownMenuTrigger asChild>
   388→                      <Button variant="ghost" size="icon" aria-label="More actions" className="focus-glow">
   389→                        <MoreHorizontal className="h-4 w-4" />
   390→                      </Button>
   391→                    </DropdownMenuTrigger>
   392→                    <DropdownMenuContent align="end" className="w-48">
   393→                      <DropdownMenuItem
   394→                        onClick={toggleBlock}
   395→                        disabled={blockPending}
   396→                        className={cn("cursor-pointer", !blockedByMe && "text-rose-300 focus:text-rose-300")}
   397→                      >
   398→                        {blockPending ? (
   399→                          <Loader2 className="h-4 w-4 animate-spin" />
   400→                        ) : (
   401→                          <Ban className="h-4 w-4" />
   402→                        )}
   403→                        {blockedByMe ? "Unblock user" : "Block user"}
   404→                      </DropdownMenuItem>
   405→                      <DropdownMenuSeparator />
   406→                      <DropdownMenuItem
   407→                        onClick={() => setReportOpen(true)}
   408→                        className="cursor-pointer"
   409→                      >
   410→                        <Flag className="h-4 w-4" />
   411→                        Report user
   412→                      </DropdownMenuItem>
   413→                    </DropdownMenuContent>
   414→                  </DropdownMenu>
   415→                </>
   416→              )}
   417→            </div>
   418→          </div>
   419→        </motion.section>
   420→
   421→        {/* ── Stats ──────────────────────────────────────────── */}
   422→        <div className="grid grid-cols-2 gap-4 md:grid-cols-4">
   423→          <StatCard label="Followers" value={profile.followerCount} />
   424→          <StatCard label="Following" value={profile.followingCount} />
   425→          <StatCard label="Projects" value={profile.projectCount} />
   426→          <StatCard label="Reputation" value={profile.reputation ?? 0} />
   427→        </div>
   428→
   429→        {/* ── Tabs ───────────────────────────────────────────── */}
   430→        <Tabs defaultValue="projects">
   431→          <TabsList className="w-full justify-start overflow-x-auto sm:w-fit">
   432→            <TabsTrigger value="projects">Projects</TabsTrigger>
   433→            <TabsTrigger value="contributions">Contributions</TabsTrigger>
   434→            <TabsTrigger value="about">About</TabsTrigger>
   435→          </TabsList>
   436→
   437→          {/* Projects */}
   438→          <TabsContent value="projects" className="mt-5">
   439→            {data && data.projects.length > 0 ? (
   440→              <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   441→                {data.projects.map((p) => (
   442→                  <ProjectCardView key={p.id} project={p} />
   443→                ))}
   444→              </div>
   445→            ) : (
   446→              <EmptyState
   447→                icon={Globe}
   448→                title="No public projects yet"
   449→                description={`When ${profile.displayName} publishes projects, they'll show up here.`}
   450→              />
   451→            )}
   452→          </TabsContent>
   453→
   454→          {/* Contributions */}
   455→          <TabsContent value="contributions" className="mt-5">
   456→            <div className="mb-5 flex items-start gap-2.5 rounded-xl border border-white/[0.06] bg-white/[0.03] px-4 py-3">
   457→              <Info className="mt-0.5 h-4 w-4 shrink-0 text-cyan-300" />
   458→              <p className="text-sm text-muted-foreground">
   459→                Reputation is earned through verified contributions only — it can&apos;t be bought.
   460→              </p>
   461→            </div>
   462→
   463→            {data && data.contributions.length > 0 ? (
   464→              <ol className="relative ml-3 space-y-6 border-l border-white/[0.08] pl-7">
   465→                {data.contributions.map((c) => {
   466→                  const meta = CONTRIBUTION_META[c.eventType] ?? {
   467→                    icon: Sparkles,
   468→                    label: "Contributed",
   469→                  };
   470→                  const Icon = meta.icon;
   471→                  return (
   472→                    <li key={c.id} className="relative">
   473→                      <span className="absolute -left-10 top-0 flex h-6 w-6 items-center justify-center rounded-full border border-white/10 bg-[#0e0e16]">
   474→                        <Icon className="h-3 w-3 text-cyan-300" />
   475→                      </span>
   476→                      <div className="flex flex-wrap items-center gap-x-2.5 gap-y-1.5">
   477→                        <p className="text-sm">
   478→                          {meta.label}
   479→                          {c.project && (
   480→                            <>
   481→                              <span className="text-muted-foreground"> · </span>
   482→                              <button
   483→                                onClick={() =>
   484→                                  c.project && navigate({ name: "project", slug: c.project.slug })
   485→                                }
   486→                                className="rounded font-medium hover:text-cyan-300 focus-glow"
   487→                              >
   488→                                {c.project.name}
   489→                              </button>
   490→                            </>
   491→                          )}
   492→                        </p>
   493→                        <span className="rounded-md bg-violet-500/10 px-1.5 py-0.5 font-mono text-[10px] font-medium text-violet-300">
   494→                          +{c.xp} XP
   495→                        </span>
   496→                        {c.reputation > 0 && (
   497→                          <span className="rounded-md bg-cyan-500/10 px-1.5 py-0.5 font-mono text-[10px] font-medium text-cyan-300">
   498→                            +{c.reputation} REP
   499→                          </span>
   500→                        )}
   501→                        <span className="ml-auto text-xs text-muted-foreground">
   502→                          {formatDistanceToNow(new Date(c.createdAt), { addSuffix: true })}
   503→                        </span>
   504→                      </div>
   505→                    </li>
   506→                  );
   507→                })}
   508→              </ol>
   509→            ) : (
   510→              <EmptyState
   511→                icon={Flag}
   512→                title="No contributions yet"
   513→                description="Verified tasks, milestones, releases, and collaborations will appear here as they happen."
   514→              />
   515→            )}
   516→          </TabsContent>
   517→
   518→          {/* About */}
   519→          <TabsContent value="about" className="mt-5">
   520→            {!profile.bio && skillNames.length === 0 && goalNames.length === 0 && interestNames.length === 0 && !websiteHost ? (
   521→              <EmptyState
   522→                icon={UserRound}
   523→                title="This creator hasn't added details yet"
   524→                description="Their bio, skills, and interests will appear here once they complete their profile."
   525→              />
   526→            ) : (
   527→              <div className="grid gap-4 md:grid-cols-2">
   528→                <div className="glass-card rounded-xl p-5 md:col-span-2">
   529→                  <h3 className="font-mono text-[10px] uppercase tracking-widest text-muted-foreground">
   530→                    Bio
   531→                  </h3>
   532→                  <p className="mt-2 whitespace-pre-line text-sm leading-relaxed text-foreground/85">
   533→                    {profile.bio || "No bio yet."}
   534→                  </p>
   535→                </div>
   536→
   537→                <div className="glass-card rounded-xl p-5">
   538→                  <h3 className="font-mono text-[10px] uppercase tracking-widest text-muted-foreground">
   539→                    Skills
   540→                  </h3>
   541→                  {skillNames.length > 0 ? (
   542→                    <div className="mt-3 flex flex-wrap gap-2">
   543→                      {skillNames.map((s) => (
   544→                        <span
   545→                          key={s}
   546→                          className="rounded-full border border-violet-500/25 bg-violet-500/10 px-2.5 py-1 text-xs text-violet-300"
   547→                        >
   548→                          {s}
   549→                        </span>
   550→                      ))}
