    90→function useAcademyData(user: SessionUser | null) {
    91→  const [progress, setProgress] = useState<ProgressMap>({});
    92→  const [certs, setCerts] = useState<CertRow[]>([]);
    93→  const [leaders, setLeaders] = useState<LeaderRow[] | null>(null);
    94→  const [loaded, setLoaded] = useState(false);
    95→
    96→  const refresh = useCallback(async () => {
    97→    if (!user) {
    98→      setProgress({});
    99→      setCerts([]);
   100→      setLoaded(true);
   101→      return;
   102→    }
   103→    try {
   104→      const res = await get<{ progress: { courseId: string; completedLessons: string[]; lastLessonId: string | null }[]; certificates: CertRow[] }>("/api/academy");
   105→      const map: ProgressMap = {};
   106→      for (const p of res.progress) map[p.courseId] = { done: p.completedLessons, last: p.lastLessonId };
   107→      setProgress(map);
   108→      setCerts(res.certificates);
   109→    } catch {
   110→      /* offline — keep defaults */
   111→    } finally {
   112→      setLoaded(true);
   113→    }
   114→  }, [user]);
   115→
   116→  useEffect(() => {
   117→    void refresh();
   118→  }, [refresh]);
   119→
   120→  useEffect(() => {
   121→    if (!user) {
   122→      setLeaders(null);
   123→      return;
   124→    }
   125→    get<{ rows: LeaderRow[] }>("/api/academy/leaderboard")
   126→      .then((r) => setLeaders(r.rows))
   127→      .catch(() => setLeaders([]));
   128→  }, [user]);
   129→
   130→  return { progress, certs, leaders, loaded, refresh, setProgress, setCerts };
   131→}

   135→function ProgressRing({ value, size = 56 }: { value: number; size?: number }) {
   136→  const stroke = 5;
   137→  const r = (size - stroke) / 2;
   138→  const c = 2 * Math.PI * r;
   139→  return (
   140→    <svg width={size} height={size} className="shrink-0 -rotate-90" role="img" aria-label={`${Math.round(value * 100)}%`}>
   141→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.08)" strokeWidth={stroke} />
   142→      <motion.circle
   143→        cx={size / 2}
   144→        cy={size / 2}
   145→        r={r}
   146→        fill="none"
   147→        stroke="url(#ringGrad)"
   148→        strokeWidth={stroke}
   149→        strokeLinecap="round"
   150→        strokeDasharray={c}
   151→        initial={{ strokeDashoffset: c }}
   152→        animate={{ strokeDashoffset: c * (1 - value) }}
   153→        transition={{ duration: 1.1, ease: EASE }}
   154→      />
   155→      <defs>
   156→        <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
   157→          <stop offset="0%" stopColor="#22d3ee" />
   158→          <stop offset="100%" stopColor="#a78bfa" />
   159→        </linearGradient>
   160→      </defs>
   161→    </svg>
   162→  );
   163→}

   444→function CourseView({ course, initialLessonId, progress, onBack, onNavigateLesson, onChallengePassed, t }: {
   445→  course: Course;
   446→  initialLessonId?: string;
   447→  progress: ProgressMap;
   448→  onBack: () => void;
   449→  onNavigateLesson: (courseId: string, lessonId: string) => void;
   450→  onChallengePassed: (courseId: string, lessonId: string) => void;
   451→  t: ReturnType<typeof useT>;
   452→}) {
   453→  const locale = useLocale();
   454→  const [tab, setTab] = useState<"course" | "playground">("course");
   455→  const done = progress[course.id]?.done ?? [];
   456→  const doneSet = useMemo(() => new Set(done), [done]);
   457→
   458→  const flat = useMemo(
   459→    () => course.modules.flatMap((m, mi) => m.lessons.map((l, li) => ({ lesson: l, mi, li }))),
   460→    [course],
   461→  );
   462→
   463→  const current = (() => {
   464→    if (initialLessonId) return flat.find((f) => f.lesson.id === initialLessonId) ?? null;
   465→    const last = progress[course.id]?.last;
   466→    if (last) {
   467→      const found = flat.find((f) => f.lesson.id === last);
   468→      if (found) return found;
   469→    }
   470→    const next = flat.find((f) => !doneSet.has(f.lesson.id));
   471→    return next ?? flat[0] ?? null;
   472→  })();
   473→
   474→  const accent = course.accent === "cyan" ? "text-cyan-300" : "text-amber-300";
   475→

   481→          <div className="flex items-center gap-3">
   482→            <span className="grid h-11 w-11 place-items-center rounded-2xl bg-white/[0.05] text-2xl ring-1 ring-inset ring-white/10" aria-hidden="true">
   483→              {course.emoji}
   484→            </span>
   485→            <div>
   486→              <h1 className="text-xl font-bold text-white sm:text-2xl">{pick(course.title, locale)}</h1>

   488→                {t("academy.lessonsDone", { done: done.length, total: flat.length })}

   509→                    transition={{ type: "spring", stiffness: 420, damping: 34 }}
   510→                    aria-hidden="true"

   512→                )}
   513→                <span className="relative">{tb === "course" ? "📚 Course" : "🛝 Playground"}</span>

   515→            ))}

   517→        </div>
   518→      </FadeIn>

   522→          <motion.div

   524→            initial={{ opacity: 0, y: 12 }}

   529→            <FadeIn>

   532→                  <Sparkles className="h-4 w-4 text-cyan-300" />

   535→                <p className="mb-4 mt-1 text-sm text-muted-foreground">{t("academy.playgroundHint")}</p>

   540→                )}
   541→              </div>
   542→            </FadeIn>
   543→          </motion.div>
   544→        ) : (
   545→          <motion.div
   546→            key="course"
   547→            initial={{ opacity: 0, y: 12 }}
   548→            animate={{ opacity: 1, y: 0 }}
   549→            exit={{ opacity: 0, y: -8, transition: { duration: 0.14 } }}
   550→            transition={{ duration: 0.3, ease: EASE }}
   551→            className="grid gap-5 lg:grid-cols-[290px_minmax(0,1fr)]"
   552→          >
   553→            {/* sidebar (desktop) */}
   554→            <aside className="hidden lg:block">
   555→              <div className="sticky top-20 max-h-[calc(100vh-6rem)] overflow-y-auto rounded-2xl border border-white/10 bg-white/[0.02] p-3 academy-scroll">
   556→                <ModuleList course={course} doneSet={doneSet} currentId={current?.lesson.id} onNavigateLesson={onNavigateLesson} t={t} />
   557→              </div>
   558→            </aside>
   559→
   560→            {/* mobile lesson picker */}
   561→            <details className="lg:hidden rounded-2xl border border-white/10 bg-white/[0.02] p-3">
   562→              <summary className="flex min-h-[44px] cursor-pointer items-center justify-between rounded-xl px-3 text-sm font-semibold text-white focus-glow">
   563→                <span>{current ? pick(current.lesson.title, locale) : "Lessons"}</span>
   564→                <ChevronDown className="h-4 w-4 text-muted-foreground" aria-hidden="true" />

   566→              <div className="mt-2 max-h-80 overflow-y-auto academy-scroll">
   567→                <ModuleList course={course} doneSet={doneSet} currentId={current?.lesson.id} onNavigateLesson={(cid, lid) => { onNavigateLesson(cid, lid); }} t={t} />
   568→              </div>
   569→            </details>
   570→
   571→            {/* player */}
   572→            <div className="min-w-0">

   575→                  key={current.lesson.id}

   577→                  lesson={current.lesson}

   579→                  lessonIndex={current.li}
   580→                  completed={doneSet.has(current.lesson.id)}
   581→                  onChallengePassed={() => onChallengePassed(course.id, current.lesson.id)}

   583→                  onBack={onBack}

   585→                />
   586→              ) : (
   587→                <div className="grid place-items-center rounded-2xl border border-dashed border-white/10 p-10 text-center">

   589→                  <p className="mt-3 text-sm text-muted-foreground">{t("academy.hubSubtitle")}</p>
   590→                </div>
   591→              )}
   592→            </div>
   593→          </motion.div>
   594→        )}
   595→      </AnimatePresence>
   596→    </div>
   597→  );
   598→}
   599→
   600→function ModuleList({ course, doneSet, currentId, onNavigateLesson, t }: {
   601→  course: Course;
   602→  doneSet: Set<string>;
   603→  currentId?: string;
   604→  onNavigateLesson: (courseId: string, lessonId: string) => void;
   605→  t: ReturnType<typeof useT>;
   606→}) {
   607→  const locale = useLocale();
   608→  const [open, setOpen] = useState<Record<string, boolean>>(() => {
   609→    const initial: Record<string, boolean> = {};
   610→    // open the module containing the current lesson
   611→    for (const m of course.modules) {
   612→      if (m.lessons.some((l) => l.id === currentId)) initial[m.id] = true;
   613→    }
   614→    return initial;
   615→  });
   616→
   617→  return (
   618→    <nav className="space-y-2" aria-label={t("academy.hubTitle")}>
   619→      {course.modules.map((m, mi) => {
   620→        const isOpen = open[m.id] ?? false;
   621→        const moduleDone = m.lessons.filter((l) => doneSet.has(l.id)).length;
   622→        return (
   623→          <div key={m.id} className="rounded-xl border border-white/[0.06] bg-white/[0.015]">
   624→            <button
   625→              onClick={() => setOpen((o) => ({ ...o, [m.id]: !isOpen }))}
   626→              aria-expanded={isOpen}
   627→              className="flex min-h-[48px] w-full items-center gap-2.5 px-3 py-2 text-start focus-glow"
   628→            >
   629→              <span className={cn("grid h-6 w-6 shrink-0 place-items-center rounded-md text-[11px] font-bold", moduleDone === m.lessons.length ? "bg-emerald-500/20 text-emerald-300" : "bg-white/[0.06] text-muted-foreground")}>
   630→                {moduleDone === m.lessons.length ? "✓" : mi + 1}
   631→              </span>
   632→              <span className="min-w-0 flex-1 truncate text-sm font-medium text-slate-200">{pick(m.title, locale)}</span>
   633→              <ChevronDown className={cn("h-4 w-4 shrink-0 text-muted-foreground transition-transform", isOpen && "rotate-180")} aria-hidden="true" />
   634→            </button>
   635→            <AnimatePresence initial={false}>
   636→              {isOpen && (
   637→                <motion.ul
   638→                  initial={{ height: 0, opacity: 0 }}
   639→                  animate={{ height: "auto", opacity: 1 }}
   640→                  exit={{ height: 0, opacity: 0 }}
   641→                  transition={{ duration: 0.25, ease: EASE }}
   642→                  className="overflow-hidden pb-1.5"
   643→                >
   644→                  {m.lessons.map((l, li) => {
   645→                    const isDone = doneSet.has(l.id);
   646→                    const isCurrent = l.id === currentId;
   647→                    return (
   648→                      <li key={l.id}>
   649→                        <button
   650→                          onClick={() => onNavigateLesson(course.id, l.id)}
   651→                          aria-current={isCurrent ? "page" : undefined}
   652→                          className={cn(
   653→                            "flex min-h-[44px] w-full items-center gap-2.5 rounded-lg px-3 py-1.5 text-start text-[13px] transition focus-glow",
   654→                            isCurrent ? "bg-cyan-400/10 text-cyan-200 ring-1 ring-inset ring-cyan-400/25" : "text-muted-foreground hover:bg-white/[0.04] hover:text-foreground",
   655→                          )}
   656→                        >
   657→                          <span
   658→                            className={cn(
   659→                              "grid h-[18px] w-[18px] shrink-0 place-items-center rounded-full text-[9px] font-bold",
   660→                              isDone ? "bg-emerald-500/25 text-emerald-300" : "bg-white/[0.07] text-slate-400",
   661→                            )}
   662→                            aria-hidden="true"
   663→                          >
   664→                            {isDone ? "✓" : li + 1}
   665→                          </span>
   666→                          <span className="min-w-0 flex-1 truncate">{pick(l.title, locale)}</span>
   667→                          <span className="shrink-0 text-[10px] text-muted-foreground/60">{l.minutes}{t("academy.minutesShort").slice(0, 1)}</span>
   668→                        </button>
   669→                      </li>
   670→                    );
   671→                  })}
   672→                </motion.ul>
   673→              )}
   674→            </AnimatePresence>
   675→          </div>
   676→        );
   677→      })}
   678→    </nav>
   679→  );
   680→}

   684→export function AcademyView({ courseSlug, lessonId }: { courseSlug?: string; lessonId?: string }) {
   685→  const t = useT();
   686→  const user = useApp((s) => s.user);
   687→  const navigate = useApp((s) => s.navigate);
   688→  const setUser = useApp((s) => s.setUser);
   689→  const { progress, certs, leaders, setProgress, refresh } = useAcademyData(user);
   690→  const [certDialog, setCertDialog] = useState<CertInfo | null>(null);
   691→  const course = useMemo(() => COURSES.find((c) => c.id === courseSlug), [courseSlug]);
   692→
   693→  const handleChallengePassed = useCallback(
   694→    async (courseId: string, lessonId: string) => {
   695→      if (!user) {
   696→        toast.info(t("academy.loginPrompt"));
   697→        return;
   698→      }
   699→      try {
   700→        const res = await post<{
   701→          added: boolean;
   702→          xpAwarded: number;
   703→          completedLessons: string[];
   704→          courseCompleted: boolean;
   705→          certificate: { code: string; createdAt: string } | null;
   706→          user: { xp: number; reputation: number };
   707→        }>("/api/academy/lessons", { courseId, lessonId });
   708→
   709→        setProgress((p) => ({
   710→          ...p,
   711→          [courseId]: { done: res.completedLessons, last: lessonId },
   712→        }));
   713→        setUser({ ...user, xp: res.user.xp, reputation: res.user.reputation });
   714→
   715→        if (res.added && res.xpAwarded > 0) {
   716→          toast.success(t("academy.lessonDoneToast", { xp: res.xpAwarded }));
   717→        }
   718→        if (res.courseCompleted && res.certificate) {
   719→          const c = COURSES.find((x) => x.id === courseId);
   720→          toast.success(t("academy.courseDoneToast", { xp: res.xpAwarded }), { duration: 6000 });
   721→          setCertDialog({
   722→            courseTitle: c ? pick(c.title, "en") : courseId,
   723→            recipient: user.displayName,
   724→            code: res.certificate.code,
   725→            issuedAt: new Date(res.certificate.createdAt).toLocaleDateString(),
   726→            emoji: c?.emoji ?? "🏆",
   727→          });
   728→          void refresh();
   729→        }
   730→      } catch (e) {
   731→        toast.error(e instanceof Error ? e.message : t("common.error"));
   732→      }
   733→    },
   734→    [user, t, setProgress, setUser, refresh],
   735→  );
   736→
   737→  const showCert = useCallback(
   738→    (courseId: string, code: string, createdAt: string) => {
   739→      const c = COURSES.find((x) => x.id === courseId);
   740→      setCertDialog({
   741→        courseTitle: c ? pick(c.title, "en") : courseId,
   742→        recipient: user?.displayName ?? "TEKA Learner",
   743→        code,
   744→        issuedAt: new Date(createdAt).toLocaleDateString(),
   745→        emoji: c?.emoji ?? "🏆",
   746→      });
   747→    },
   748→    [user],
   749→  );
   750→
   751→  return (
   752→    <main className="mx-auto w-full max-w-6xl flex-1 px-4 py-6 sm:px-6 sm:py-8">
   753→      {course ? (
   754→        <CourseView
   755→          course={course}
   756→          initialLessonId={lessonId}
   757→          progress={progress}
   758→          onBack={() => navigate({ name: "academy" })}
   759→          onNavigateLesson={(cid, lid) => navigate({ name: "academy", course: cid, lesson: lid })}
   760→          onChallengePassed={handleChallengePassed}
   761→          t={t}
   762→        />
   763→      ) : (
   764→        <Hub
   765→          onOpenCourse={(cid) => navigate({ name: "academy", course: cid })}
   766→          progress={progress}
   767→          certs={certs}
   768→          leaders={leaders}
   769→          onShowCert={showCert}
   770→        />
   771→      )}
   772→
   773→      <CertificateDialog
   774→        cert={certDialog}
   775→        open={!!certDialog}
   776→        onClose={() => setCertDialog(null)}
   777→        t={t}
   778→      />
   779→    </main>
   780→  );
   781→}
