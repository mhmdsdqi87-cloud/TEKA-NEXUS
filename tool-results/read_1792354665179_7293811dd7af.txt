  1016→function CommentsSection({
  1017→  projectSlug,
  1018→  comments,
  1019→  user,
  1020→  canManage,
  1021→  setComments,
  1022→  refetchComments,
  1023→}: {
  1024→  projectSlug: string;
  1025→  comments: CommentDTO[] | null;
  1026→  user: ReturnType<typeof useApp.getState>["user"];
  1027→  canManage: boolean;
  1028→  setComments: React.Dispatch<React.SetStateAction<CommentDTO[] | null>>;
  1029→  refetchComments: () => Promise<void>;
  1030→}) {
  1031→  const [body, setBody] = useState("");
  1032→  const [sending, setSending] = useState(false);
  1033→  const [replyTo, setReplyTo] = useState<string | null>(null);
  1034→  const [replyBody, setReplyBody] = useState("");
  1035→  const [deleteId, setDeleteId] = useState<string | null>(null);
  1036→  const navigate = useApp((s) => s.navigate);
  1037→
  1038→  async function submit() {
  1039→    if (!body.trim()) return;
  1040→    setSending(true);
  1041→    try {
  1042→      const r = await post<{ comment: CommentDTO }>(`/api/projects/${enc(projectSlug)}/comments`, {
  1043→        body,
  1044→      });
  1045→      setComments((prev) => [r.comment, ...(prev ?? [])]);
  1046→      setBody("");
  1047→      toast.success("Comment posted");
  1048→    } catch (e) {
  1049→      toast.error(errText(e));
  1050→    } finally {
  1051→      setSending(false);
  1052→    }
  1053→  }
  1054→
  1055→  async function submitReply() {
  1056→    if (!replyTo || !replyBody.trim()) return;
  1057→    setSending(true);
  1058→    try {
  1059→      await post(`/api/projects/${enc(projectSlug)}/comments`, {
  1060→        body: replyBody,
  1061→        parentId: replyTo,
  1062→      });
  1063→      setReplyBody("");
  1064→      setReplyTo(null);
  1065→      await refetchComments();
  1066→      toast.success("Reply posted");
  1067→    } catch (e) {
  1068→      toast.error(errText(e));
  1069→    } finally {
  1070→      setSending(false);
  1071→    }
  1072→  }
  1073→
  1074→  async function confirmDelete() {
  1075→    if (!deleteId) return;
  1076→    try {
  1077→      await del(`/api/comments/${deleteId}`);
  1078→      await refetchComments();
  1079→      toast.success("Comment deleted");
  1080→    } catch (e) {
  1081→      toast.error(errText(e));
  1082→    } finally {
  1083→      setDeleteId(null);
  1084→    }
  1085→  }
  1086→
  1087→  return (
  1088→    <section className="glass-card rounded-2xl p-5 md:p-6" aria-label="Project discussion">
  1089→      <h2 className="font-display mb-4 text-lg font-semibold">
  1090→        Discussion{" "}
  1091→        <span className="ml-1 text-sm font-normal text-muted-foreground">
  1092→          ({comments?.length ?? 0})
  1093→        </span>
  1094→      </h2>
  1095→
  1096→      {user ? (
  1097→        <div className="mb-6">
  1098→          <Textarea
  1099→            value={body}
  1100→            onChange={(e) => setBody(e.target.value)}
  1101→            placeholder="Share your thoughts on this project..."
  1102→            rows={3}
  1103→            maxLength={LIMITS.comment}
  1104→            className="focus-glow resize-none bg-white/[0.03]"
  1105→          />
  1106→          <div className="mt-2 flex items-center justify-between">
  1107→            <span className="font-mono text-[11px] text-muted-foreground">
  1108→              {body.length}/{LIMITS.comment}
  1109→            </span>
  1110→            <Button size="sm" className="btn-hero" onClick={submit} disabled={sending || !body.trim()}>
  1111→              {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
  1112→              Comment
  1113→            </Button>
  1114→          </div>
  1115→        </div>
  1116→      ) : (
  1117→        <div className="mb-6 flex flex-col items-center gap-2 rounded-xl border border-white/[0.07] bg-white/[0.02] py-5">
  1118→          <p className="text-sm text-muted-foreground">Sign in to join the discussion</p>
  1119→          <Button size="sm" variant="outline" onClick={() => navigate({ name: "login" })}>
  1120→            Sign in
  1121→          </Button>
  1122→        </div>
  1123→      )}
  1124→
  1125→      {comments === null ? (
  1126→        <ListRowsSkeleton rows={2} />
  1127→      ) : comments.length === 0 ? (
  1128→        <p className="py-6 text-center text-sm text-muted-foreground">
  1129→          No comments yet. Be the first to say something.
  1130→        </p>
  1131→      ) : (
  1132→        <ul className="space-y-4">
  1133→          {comments.map((c) => (
  1134→            <li key={c.id}>
  1135→              <CommentItem
  1136→                comment={c}
  1137→                user={user}
  1138→                canManage={canManage}

  1161→              {replyTo === c.id && (

  1180→                    >
  1181→                      Cancel

  1183→                    <Button
  1184→                      size="sm"

  1186→                      onClick={submitReply}
  1187→                      disabled={sending || !replyBody.trim()}
