    65→function UnreadBadge({ count, className }: { count: number; className?: string }) {
    66→  const t = useT();
    67→  if (count <= 0) return null;

    69→    <span
    70→      className={cn(
    71→        "inline-flex min-w-[18px] h-[18px] px-1 items-center justify-center rounded-full bg-cyan-500/20 border border-cyan-400/30 text-cyan-300 text-[10px] font-semibold leading-none",
    72→        className
    73→      )}
    74→      aria-label={t("nav.unreadNotificationsAria", { count })}
    75→    >
    76→      {/* keyed by count so every change replays the spring pop */}
    77→      <motion.span
    78→        key={count}
    79→        aria-hidden="true"
    80→        initial={{ scale: 0.4, opacity: 0 }}
    81→        animate={{ scale: 1, opacity: 1 }}
    82→        transition={SPRING}
    83→        className="inline-flex items-center"
    84→      >
    85→        {count > 99 ? "99+" : count}

   427→// ─── Global search dialog ────────────────────────────────────
   428→export function SearchDialog() {
   429→  const open = useApp((s) => s.searchOpen);
   430→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   431→  const navigate = useApp((s) => s.navigate);
   432→
   433→  const t = useT();
   434→  const [query, setQuery] = useState("");
   435→  const [data, setData] = useState<{ q: string; results: SearchResults } | null>(null);
   436→  const reqId = useRef(0);
   437→
   438→  // Debounced server search (250ms, min 2 chars).
   439→  useEffect(() => {
   440→    const q = query.trim();
   441→    if (q.length < 2) return;
   442→    const id = ++reqId.current;
   443→    const timer = setTimeout(async () => {
   444→      try {
   445→        const res = await get<SearchResults>(`/api/search${qs({ q })}`);
   446→        if (reqId.current === id) setData({ q, results: res });
   447→      } catch {
   448→        if (reqId.current === id) setData({ q, results: { projects: [], users: [], tags: [] } });
   449→      }
   450→    }, 250);
   451→    return () => clearTimeout(timer);
   452→  }, [query]);
   453→
   454→  // Close + reset (event handler, not an effect).
   455→  const handleOpenChange = (next: boolean) => {
   456→    setSearchOpen(next);
   457→    if (!next) {
   458→      reqId.current++;
   459→      setQuery("");
   460→      setData(null);
   461→    }
   462→  };
   463→
   464→  const q = query.trim();
   465→  // Results are only shown when they belong to the current query;
   466→  // a newer query with no response yet renders the "Searching…" state.
   467→  const results = data && data.q === q ? data.results : null;
   468→  const searching = q.length >= 2 && results === null;
   469→  const hasResults =
   470→    results && (results.projects.length > 0 || results.users.length > 0 || results.tags.length > 0);
   471→
   472→  return (
   473→    <CommandDialog
   474→      open={open}
   475→      onOpenChange={handleOpenChange}
   476→      title={t("search.title")}
   477→      description={t("search.description")}
   478→      showCloseButton={false}
   479→      className="max-w-xl top-[15%] translate-y-0"
   480→    >
   481→      <CommandInput
   482→        value={query}
   483→        onValueChange={setQuery}
   484→        placeholder={t("search.placeholder")}
   485→        aria-label={t("search.inputAria")}
   486→      />
   487→      <CommandList>
   488→        {q.length < 2 ? (
   489→          <div className="py-8 px-4 text-center text-sm text-muted-foreground">
   490→            {t("search.minChars")}
   491→          </div>
   492→        ) : searching ? (
   493→          <div className="py-8 px-4 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   494→            <Loader2 className="h-4 w-4 animate-spin" /> {t("search.searching")}
   495→          </div>
   496→        ) : hasResults ? (
   497→          <>
   498→            {results!.projects.length > 0 && (
   499→              <CommandGroup heading={t("common.projects")}>
   500→                {results!.projects.map((p) => (
   501→                  <CommandItem
   502→                    key={p.id}
   503→                    value={`${q} ${p.name}`}
   504→                    onSelect={() => {
   505→                      setSearchOpen(false);
   506→                      navigate({ name: "project", slug: p.slug });
   507→                    }}
   508→                  >
   509→                    <FileCode2 className="text-cyan-300" />
   510→                    <span className="truncate">{p.name}</span>
   511→                    <span className="ml-auto truncate max-w-[45%] text-xs text-muted-foreground">{p.tagline}</span>
   512→                  </CommandItem>
   513→                ))}
   514→              </CommandGroup>
   515→            )}
   516→            {results!.users.length > 0 && (
   517→              <CommandGroup heading={t("search.usersHeading")}>
   518→                {results!.users.map((u) => (
   519→                  <CommandItem
   520→                    key={u.id}
   521→                    value={`${q} ${u.displayName} ${u.username}`}
   522→                    onSelect={() => {
   523→                      setSearchOpen(false);
   524→                      navigate({ name: "profile", username: u.username });
   525→                    }}
   526→                  >
   527→                    <UserAvatar user={u} size={20} />
   528→                    <span className="truncate">{u.displayName}</span>
   529→                    <span className="ml-auto text-xs text-muted-foreground">@{u.username}</span>
   530→                  </CommandItem>
   531→                ))}
   532→              </CommandGroup>
   533→            )}
   534→            {results!.tags.length > 0 && (
   535→              <CommandGroup heading={t("search.tagsHeading")}>
   536→                {results!.tags.map((tag) => (
   537→                  <CommandItem
   538→                    key={tag.tag}
   539→                    value={`${q} ${tag.tag}`}
   540→                    onSelect={() => {
   541→                      setSearchOpen(false);
   542→                      navigate({ name: "explore", query: tag.tag });
   543→                    }}
   544→                  >
   545→                    <Hash className="text-violet-300" />
   546→                    <span className="font-mono">#{tag.tag}</span>
   547→                    <span className="ml-auto text-xs text-muted-foreground">
   548→                      {t(tag.count === 1 ? "search.tagProjectCount" : "search.tagProjectCountPlural", { count: tag.count })}
   549→                    </span>
   550→                  </CommandItem>
   551→                ))}
   552→              </CommandGroup>
   553→            )}
   554→          </>
   555→        ) : (
   556→          <CommandEmpty>{t("search.noResults", { q })}</CommandEmpty>
   557→        )}
   558→      </CommandList>
   559→    </CommandDialog>
   560→  );

   711→export function AppShell({ children }: { children: React.ReactNode }) {
   712→  const user = useApp((s) => s.user);
   713→  const setUnreadCount = useApp((s) => s.setUnreadCount);
   714→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   715→  const t = useT();
   716→  const userId = user?.id ?? null;
   717→
   718→  // Poll unread notification count every 45s while signed in.
   719→  useEffect(() => {
   720→    if (!userId) return;
   721→    let cancelled = false;
   722→    const poll = async () => {
   723→      try {
   724→        const res = await get<{ notifications: unknown[]; unread: number }>("/api/notifications");
   725→        if (!cancelled) setUnreadCount(res.unread);
   726→      } catch {
   727→        /* silent — background badge poll */
   728→      }
   729→    };
   730→    poll();
   731→    const timer = setInterval(poll, 45_000);
   732→    return () => {
   733→      cancelled = true;
   734→      clearInterval(timer);
   735→    };
   736→  }, [userId, setUnreadCount]);
   737→
   738→  // "/" opens global search (ignored while typing in form fields).
   739→  useEffect(() => {
   740→    const onKey = (e: KeyboardEvent) => {
   741→      if (e.key !== "/") return;
   742→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   743→      const target = e.target as HTMLElement | null;
   744→      const tag = target?.tagName;
   745→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || target?.isContentEditable) return;
   746→      e.preventDefault();
   747→      setSearchOpen(true);
   748→    };
   749→    window.addEventListener("keydown", onKey);
   750→    return () => window.removeEventListener("keydown", onKey);
   751→  }, [setSearchOpen]);
   752→
   753→  return (
   754→    <div className="min-h-screen flex">
   755→      <LeftNav />
   756→
   757→      <div className="flex-1 flex flex-col min-w-0">
   758→        <TopBar />
   759→        <main className="flex-1 pb-24 md:pb-10" id="main-content">
   760→          <div className="max-w-5xl mx-auto px-4 md:px-8 py-6">
   761→            <motion.div
   762→              initial={{ opacity: 0, y: 8 }}
   763→              animate={{ opacity: 1, y: 0 }}
   764→              transition={{ duration: 0.3, ease: "easeOut" }}
   765→            >
   766→              {children}
   767→            </motion.div>
   768→          </div>
   769→        </main>
   770→        <footer className="py-4 text-center text-xs text-muted-foreground/70 select-none" aria-label={t("nav.footerAria")}>
   771→          {t("nav.footerTagline")}
   772→        </footer>
   773→      </div>
   774→
   775→      <RightRail />
   776→      <MobileNav />
   777→      <SearchDialog />
   778→    </div>
   779→  );
   780→}
