   391→// ─── Global search dialog ────────────────────────────────────
   392→export function SearchDialog() {
   393→  const open = useApp((s) => s.searchOpen);
   394→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   395→  const navigate = useApp((s) => s.navigate);
   396→
   397→  const t = useT();
   398→  const [query, setQuery] = useState("");
   399→  const [data, setData] = useState<{ q: string; results: SearchResults } | null>(null);
   400→  const reqId = useRef(0);
   401→
   402→  // Debounced server search (250ms, min 2 chars).
   403→  useEffect(() => {
   404→    const q = query.trim();
   405→    if (q.length < 2) return;
   406→    const id = ++reqId.current;
   407→    const timer = setTimeout(async () => {
   408→      try {
   409→        const res = await get<SearchResults>(`/api/search${qs({ q })}`);
   410→        if (reqId.current === id) setData({ q, results: res });
   411→      } catch {
   412→        if (reqId.current === id) setData({ q, results: { projects: [], users: [], tags: [] } });
   413→      }
   414→    }, 250);
   415→    return () => clearTimeout(timer);
   416→  }, [query]);
   417→
   418→  // Close + reset (event handler, not an effect).
   419→  const handleOpenChange = (next: boolean) => {
   420→    setSearchOpen(next);
   421→    if (!next) {
   422→      reqId.current++;
   423→      setQuery("");
   424→      setData(null);
   425→    }
   426→  };
   427→
   428→  const q = query.trim();
   429→  // Results are only shown when they belong to the current query;
   430→  // a newer query with no response yet renders the "Searching…" state.
   431→  const results = data && data.q === q ? data.results : null;
   432→  const searching = q.length >= 2 && results === null;
   433→  const hasResults =
   434→    results && (results.projects.length > 0 || results.users.length > 0 || results.tags.length > 0);
   435→
   436→  return (
   437→    <CommandDialog
   438→      open={open}
   439→      onOpenChange={handleOpenChange}
   440→      title={t("search.title")}
   441→      description={t("search.description")}
   442→      showCloseButton={false}
   443→      className="max-w-xl top-[15%] translate-y-0"
   444→    >
   445→      <CommandInput
   446→        value={query}
   447→        onValueChange={setQuery}
   448→        placeholder={t("search.placeholder")}
   449→        aria-label={t("search.inputAria")}
   450→      />
   451→      <CommandList>
   452→        {q.length < 2 ? (
   453→          <div className="py-8 px-4 text-center text-sm text-muted-foreground">
   454→            {t("search.minChars")}
   455→          </div>
   456→        ) : searching ? (
   457→          <div className="py-8 px-4 flex items-center justify-center gap-2 text-sm text-muted-foreground">
   458→            <Loader2 className="h-4 w-4 animate-spin" /> {t("search.searching")}
   459→          </div>
   460→        ) : hasResults ? (
   461→          <>
   462→            {results!.projects.length > 0 && (
   463→              <CommandGroup heading={t("common.projects")}>
   464→                {results!.projects.map((p) => (
   465→                  <CommandItem
   466→                    key={p.id}
   467→                    value={`${q} ${p.name}`}
   468→                    onSelect={() => {
   469→                      setSearchOpen(false);
   470→                      navigate({ name: "project", slug: p.slug });
   471→                    }}
   472→                  >
   473→                    <FileCode2 className="text-cyan-300" />
   474→                    <span className="truncate">{p.name}</span>
   475→                    <span className="ml-auto truncate max-w-[45%] text-xs text-muted-foreground">{p.tagline}</span>
   476→                  </CommandItem>
   477→                ))}
   478→              </CommandGroup>
   479→            )}
   480→            {results!.users.length > 0 && (
   481→              <CommandGroup heading={t("search.usersHeading")}>
   482→                {results!.users.map((u) => (
   483→                  <CommandItem
   484→                    key={u.id}
   485→                    value={`${q} ${u.displayName} ${u.username}`}
   486→                    onSelect={() => {
   487→                      setSearchOpen(false);
   488→                      navigate({ name: "profile", username: u.username });
   489→                    }}
   490→                  >
   491→                    <UserAvatar user={u} size={20} />
   492→                    <span className="truncate">{u.displayName}</span>
   493→                    <span className="ml-auto text-xs text-muted-foreground">@{u.username}</span>
   494→                  </CommandItem>
   495→                ))}
   496→              </CommandGroup>
   497→            )}
   498→            {results!.tags.length > 0 && (
   499→              <CommandGroup heading={t("search.tagsHeading")}>
   500→                {results!.tags.map((tag) => (
   501→                  <CommandItem
   502→                    key={tag.tag}
   503→                    value={`${q} ${tag.tag}`}
   504→                    onSelect={() => {
   505→                      setSearchOpen(false);
   506→                      navigate({ name: "explore", query: tag.tag });
   507→                    }}
   508→                  >
   509→                    <Hash className="text-violet-300" />
   510→                    <span className="font-mono">#{tag.tag}</span>
   511→                    <span className="ml-auto text-xs text-muted-foreground">
   512→                      {t(tag.count === 1 ? "search.tagProjectCount" : "search.tagProjectCountPlural", { count: tag.count })}
   513→                    </span>
   514→                  </CommandItem>
   515→                ))}
   516→              </CommandGroup>
   517→            )}
   518→          </>
   519→        ) : (
   520→          <CommandEmpty>{t("search.noResults", { q })}</CommandEmpty>
   521→        )}
   522→      </CommandList>
   523→    </CommandDialog>
   524→  );
   525→}

   674→// ─── Shell ───────────────────────────────────────────────────
   675→export function AppShell({ children }: { children: React.ReactNode }) {
   676→  const user = useApp((s) => s.user);
   677→  const setUnreadCount = useApp((s) => s.setUnreadCount);
   678→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   679→  const t = useT();
   680→  const userId = user?.id ?? null;
   681→
   682→  // Poll unread notification count every 45s while signed in.
   683→  useEffect(() => {
   684→    if (!userId) return;
   685→    let cancelled = false;
   686→    const poll = async () => {
   687→      try {
   688→        const res = await get<{ notifications: unknown[]; unread: number }>("/api/notifications");
   689→        if (!cancelled) setUnreadCount(res.unread);
   690→      } catch {
   691→        /* silent — background badge poll */
   692→      }
   693→    };
   694→    poll();
   695→    const timer = setInterval(poll, 45_000);
   696→    return () => {
   697→      cancelled = true;
   698→      clearInterval(timer);
   699→    };
   700→  }, [userId, setUnreadCount]);
   701→
   702→  // "/" opens global search (ignored while typing in form fields).
   703→  useEffect(() => {
   704→    const onKey = (e: KeyboardEvent) => {
   705→      if (e.key !== "/") return;
   706→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   707→      const target = e.target as HTMLElement | null;
   708→      const tag = target?.tagName;
   709→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || target?.isContentEditable) return;
   710→      e.preventDefault();
   711→      setSearchOpen(true);
   712→    };
   713→    window.addEventListener("keydown", onKey);
   714→    return () => window.removeEventListener("keydown", onKey);
   715→  }, [setSearchOpen]);
   716→
   717→  return (
   718→    <MotionConfig reducedMotion="user">
   719→      <div className="min-h-screen flex">
   720→        <LeftNav />
   721→
   722→        <div className="flex-1 flex flex-col min-w-0">
   723→          <TopBar />
   724→          <main className="flex-1 pb-24 md:pb-10" id="main-content">
   725→            <div className="max-w-5xl mx-auto px-4 md:px-8 py-6">
   726→              <motion.div
   727→                initial={{ opacity: 0, y: 8 }}
   728→                animate={{ opacity: 1, y: 0 }}
   729→                transition={{ duration: 0.3, ease: "easeOut" }}
   730→              >
   731→                {children}
   732→              </motion.div>
   733→            </div>
   734→          </main>
   735→          <footer className="py-4 text-center text-xs text-muted-foreground/70 select-none" aria-label={t("nav.footerAria")}>
   736→            {t("nav.footerTagline")}
   737→          </footer>
   738→        </div>
   739→
   740→        <RightRail />
   741→        <MobileNav />
   742→        <SearchDialog />
   743→      </div>
   744→    </MotionConfig>
   745→  );
   746→}
