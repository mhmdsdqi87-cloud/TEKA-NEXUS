   152→function ProfileTab({ user }: { user: SessionUser }) {
   153→  const setUser = useApp((s) => s.setUser);
   154→  const t = useT();
   155→  const lbl = useLabeled();
   156→
   157→  const [displayName, setDisplayName] = useState(user.displayName);
   158→  const [bio, setBio] = useState(user.bio ?? "");
   159→  const [websiteUrl, setWebsiteUrl] = useState(user.websiteUrl ?? "");
   160→  const [avatarColor, setAvatarColor] = useState(user.avatarColor);
   161→  const [skills, setSkills] = useState<string[]>([]);
   162→  const [skillsLoaded, setSkillsLoaded] = useState(false);
   163→  const [saving, setSaving] = useState(false);
   164→  const [websiteError, setWebsiteError] = useState<string | null>(null);
   165→
   166→  // SessionUser doesn't carry skills — load them from the public profile.
   167→  useEffect(() => {
   168→    let alive = true;
   169→    get<{ user: ProfileDetail }>(`/api/users/${encodeURIComponent(user.username)}`)
   170→      .then((res) => {
   171→        if (!alive) return;
   172→        setSkills(res.user.skills);
   173→        setSkillsLoaded(true);
   174→      })
   175→      .catch(() => {
   176→        if (alive) setSkillsLoaded(true);
   177→      });
   178→    return () => {
   179→      alive = false;
   180→    };
   181→  }, [user.username]);
   182→
   183→  async function saveProfile() {
   184→    const name = displayName.trim();
   185→    if (!name) {
   186→      toast.error(t("settings.errors.emptyDisplayName"));
   187→      return;
   188→    }
   189→    const site = websiteUrl.trim();
   190→    if (site && !/^https?:\/\/\S+$/i.test(site)) {
   191→      setWebsiteError(t("settings.profileTab.websiteError"));
   192→      return;
   193→    }
   194→    setWebsiteError(null);
   195→    setSaving(true);
   196→    try {
   197→      const res = await patch<{ user: SessionUser }>("/api/me/profile", {
   198→        displayName: name,
   199→        bio,
   200→        websiteUrl: site,
   201→        avatarColor,
   202→        // Only replace skills once the current set has loaded, so we never wipe them.
   203→        ...(skillsLoaded ? { skills } : {}),
   204→      });
   205→      setUser(res.user);
   206→      toast.success(t("settings.toasts.profileUpdated"));
   207→    } catch (e) {
   208→      toast.error(e instanceof Error ? e.message : t("settings.errors.saveFailed"));
   209→    } finally {
   210→      setSaving(false);
   211→    }
   212→  }
   213→
   214→  function toggleSkill(slug: string) {
   215→    setSkills((prev) => (prev.includes(slug) ? prev.filter((s) => s !== slug) : [...prev, slug]));
   216→  }
   217→
   218→  return (
   219→    <div className="space-y-5">
   220→      <section className="glass-card space-y-5 rounded-xl p-5">
   221→        <div className="space-y-2">
   222→          <Label htmlFor="display-name">{t("settings.profileTab.displayName")}</Label>
   223→          <Input
   224→            id="display-name"
   225→            value={displayName}
   226→            maxLength={60}
   227→            onChange={(e) => setDisplayName(e.target.value)}
   228→            className="focus-glow"
   229→            placeholder={t("settings.profileTab.displayNamePlaceholder")}
   230→          />
   231→        </div>
   232→
   233→        <div className="space-y-2">
   234→          <div className="flex items-center justify-between">
   235→            <Label htmlFor="bio">{t("settings.profileTab.bio")}</Label>
   236→            <span
   237→              className={cn(
   238→                "font-mono text-[10px]",
   239→                bio.length > LIMITS.bio ? "text-rose-300" : "text-muted-foreground"
   240→              )}
   241→            >
   242→              {bio.length} / {LIMITS.bio}
   243→            </span>
   244→          </div>
   245→          <Textarea
   246→            id="bio"
   247→            rows={4}
   248→            maxLength={LIMITS.bio}
   249→            value={bio}
   250→            onChange={(e) => setBio(e.target.value)}
   251→            placeholder={t("settings.profileTab.bioPlaceholder")}
   252→            className="focus-glow"
   253→          />
   254→        </div>
   255→
   256→        <div className="space-y-2">
   257→          <Label htmlFor="website">{t("settings.profileTab.website")}</Label>
   258→          <Input
   259→            id="website"
   260→            value={websiteUrl}
   261→            onChange={(e) => {
   262→              setWebsiteUrl(e.target.value);
   263→              setWebsiteError(null);
   264→            }}
   265→            placeholder="https://your-site.dev"
   266→            className={cn("font-mono text-sm focus-glow", websiteError && "border-rose-500/50")}
   267→          />
   268→          {websiteError ? (
   269→            <p className="text-xs text-rose-300">{websiteError}</p>
   270→          ) : (
   271→            <p className="text-xs text-muted-foreground">{t("settings.profileTab.websiteHint")}</p>
   272→          )}
   273→        </div>
   274→
   275→        <div className="space-y-2">
   276→          <Label>{t("settings.profileTab.avatarGradient")}</Label>
   277→          <div className="flex flex-wrap items-center gap-4">
   278→            <UserAvatar user={{ ...user, avatarColor }} size={56} />
   279→            <div className="flex flex-wrap gap-2" role="group" aria-label={t("settings.profileTab.chooseGradient")}>
   280→              {AVATAR_GRADIENT_KEYS.map((key) => (
   281→                <button
   282→                  key={key}
   283→                  type="button"
   284→                  onClick={() => setAvatarColor(key)}
   285→                  aria-label={t("settings.profileTab.gradientName", { name: key })}
   286→                  aria-pressed={avatarColor === key}
   287→                  className={cn(
   288→                    "h-9 w-9 rounded-full bg-gradient-to-br transition-all focus-glow",
   289→                    AVATAR_GRADIENTS[key],
   290→                    avatarColor === key
   291→                      ? "scale-110 ring-2 ring-cyan-300 ring-offset-2 ring-offset-[#0b0b12]"
   292→                      : "opacity-70 hover:scale-105 hover:opacity-100"
   293→                  )}
   294→                />
   295→              ))}
   296→            </div>
   297→          </div>
   298→        </div>
   299→      </section>
   300→
   301→      <section className="glass-card space-y-3 rounded-xl p-5">
   302→        <div>
   303→          <Label>{t("settings.profileTab.skills")}</Label>
   304→          <p className="mt-1 text-xs text-muted-foreground">
   305→            {t("settings.profileTab.skillsHint")}
   306→          </p>
   307→        </div>
   308→        <div className="flex flex-wrap gap-2">
   309→          {SKILLS.map((s) => {
   310→            const selected = skills.includes(s.slug);
   311→            return (
   312→              <button
   313→                key={s.slug}
   314→                type="button"
   315→                onClick={() => toggleSkill(s.slug)}
   316→                aria-pressed={selected}
   317→                className={cn(
   318→                  "rounded-full border px-3 py-1.5 text-xs transition-colors focus-glow",
   319→                  selected
   320→                    ? "border-violet-500/40 bg-violet-500/15 text-violet-200"
   321→                    : "border-white/10 bg-white/[0.03] text-muted-foreground hover:border-white/20 hover:text-foreground"
   322→                )}
   323→              >
   324→                {lbl("project.skills", s.slug, s.name)}
   325→              </button>
   326→            );
   327→          })}
   328→        </div>
   329→      </section>
   330→
   331→      <div className="flex justify-end">
   332→        <Button onClick={saveProfile} disabled={saving} className="btn-hero min-w-32">
   333→          {saving && <Loader2 className="h-4 w-4 animate-spin" />}
   334→          {t("common.saveChanges")}
   335→        </Button>
   336→      </div>
   337→    </div>
   338→  );
   339→}

   343→function AccountTab({ user }: { user: SessionUser }) {
   344→  const setUser = useApp((s) => s.setUser);
   345→  const navigate = useApp((s) => s.navigate);
   346→  const t = useT();
   347→  const locale = useI18n((s) => s.locale);
   348→  const activeLocaleNative = LOCALES.find((l) => l.code === locale)?.native ?? "English";
   349→
   350→  const [pwCurrent, setPwCurrent] = useState("");
   351→  const [pwNew, setPwNew] = useState("");
   352→  const [pwConfirm, setPwConfirm] = useState("");
   353→  const [pwPending, setPwPending] = useState(false);
   354→
   355→  const [delOpen, setDelOpen] = useState(false);
   356→  const [delStage, setDelStage] = useState<"username" | "password">("username");
   357→  const [delConfirm, setDelConfirm] = useState("");
   358→  const [delPassword, setDelPassword] = useState("");
   359→  const [delPending, setDelPending] = useState(false);
   360→  const [delError, setDelError] = useState<string | null>(null);
   361→
   362→  const pwTooShort = pwNew.length > 0 && pwNew.length < 8;
   363→  const pwMismatch = pwConfirm.length > 0 && pwNew !== pwConfirm;
   364→
   365→  async function changePassword() {
   366→    if (!pwCurrent) {
   367→      toast.error(t("settings.errors.enterCurrentPassword"));
   368→      return;
   369→    }
   370→    if (pwNew.length < 8) {
   371→      toast.error(t("settings.errors.pwTooShort"));
   372→      return;
   373→    }
   374→    if (pwNew !== pwConfirm) {
   375→      toast.error(t("settings.errors.pwMismatch"));
   376→      return;
   377→    }
   378→    setPwPending(true);
   379→    try {
   380→      await patch("/api/me/password", { currentPassword: pwCurrent, newPassword: pwNew });
   381→      toast.success(t("settings.toasts.passwordChanged"));
   382→      setPwCurrent("");
   383→      setPwNew("");
   384→      setPwConfirm("");
   385→    } catch (e) {
   386→      toast.error(e instanceof Error ? e.message : t("settings.errors.pwChangeFailed"));
   387→    } finally {
   388→      setPwPending(false);
   389→    }
   390→  }
   391→
   392→  async function deleteAccount() {
   393→    setDelPending(true);
   394→    setDelError(null);
   395→    try {
   396→      await del("/api/me/account", { password: delPassword });
   397→      setDelOpen(false);
   398→      setUser(null);
   399→      toast.success(t("settings.toasts.accountDeleted"));
   400→      navigate({ name: "landing" });
   401→    } catch (e) {
   402→      setDelError(e instanceof Error ? e.message : t("settings.errors.deleteFailed"));
   403→    } finally {
   404→      setDelPending(false);
   405→    }
   406→  }
   407→
   408→  function closeDeleteDialog(open: boolean) {
   409→    setDelOpen(open);
   410→    if (!open) {
   411→      setDelStage("username");
   412→      setDelConfirm("");
   413→      setDelPassword("");
   414→      setDelError(null);
   415→    }
   416→  }
   417→
   418→  return (
   419→    <div className="space-y-5">
   420→      <section className="glass-card space-y-4 rounded-xl p-5">
   421→        <div className="flex items-center justify-between gap-4 flex-wrap">
   422→          <div>
   423→            <h3 className="font-display font-semibold">{t("settings.accountTab.language")}</h3>
   424→            <p className="mt-0.5 text-xs text-muted-foreground">
   425→              {t("settings.accountTab.languageDescription")}
   426→            </p>
   427→          </div>
   428→          <div className="flex items-center gap-3">
   429→            <span className="hidden sm:block text-sm text-muted-foreground" aria-hidden="true">
   430→              {activeLocaleNative}
   431→            </span>
   432→            <LanguageSwitcher />
   433→          </div>
   434→        </div>
   435→      </section>
   436→
   437→      <section className="glass-card space-y-5 rounded-xl p-5">
   438→        <div className="space-y-2">
   439→          <Label htmlFor="account-email">{t("settings.accountTab.email")}</Label>
   440→          <Input
   441→            id="account-email"
   442→            value={user.email}
   443→            readOnly
   444→            className="cursor-default font-mono text-sm focus-glow"
   445→            aria-describedby="account-email-hint"
   446→          />
   447→          <p id="account-email-hint" className="text-xs text-muted-foreground">
   448→            {t("settings.accountTab.emailHint")}
   449→          </p>
   450→        </div>
   451→      </section>
   452→
   453→      <section className="glass-card space-y-4 rounded-xl p-5">
   454→        <div>
   455→          <h3 className="font-display font-semibold">{t("settings.accountTab.changePassword")}</h3>
   456→          <p className="mt-0.5 text-xs text-muted-foreground">
   457→            {t("settings.accountTab.changePasswordHint")}
   458→          </p>
   459→        </div>
   460→        <div className="grid gap-4 sm:grid-cols-3">
   461→          <div className="space-y-2">
   462→            <Label htmlFor="pw-current">{t("settings.accountTab.currentPassword")}</Label>
   463→            <Input
   464→              id="pw-current"
   465→              type="password"
   466→              autoComplete="current-password"
   467→              value={pwCurrent}
   468→              onChange={(e) => setPwCurrent(e.target.value)}
   469→              className="focus-glow"
   470→            />
   471→          </div>
   472→          <div className="space-y-2">
   473→            <Label htmlFor="pw-new">{t("settings.accountTab.newPassword")}</Label>
   474→            <Input
   475→              id="pw-new"
   476→              type="password"
   477→              autoComplete="new-password"
   478→              value={pwNew}
   479→              onChange={(e) => setPwNew(e.target.value)}
   480→              className={cn("focus-glow", pwTooShort && "border-rose-500/50")}
   481→            />
   482→            {pwTooShort && <p className="text-xs text-rose-300">{t("settings.accountTab.pwTooShort")}</p>}
   483→          </div>
   484→          <div className="space-y-2">
   485→            <Label htmlFor="pw-confirm">{t("settings.accountTab.confirmPassword")}</Label>
   486→            <Input
   487→              id="pw-confirm"
   488→              type="password"
   489→              autoComplete="new-password"
   490→              value={pwConfirm}
   491→              onChange={(e) => setPwConfirm(e.target.value)}
   492→              className={cn("focus-glow", pwMismatch && "border-rose-500/50")}
   493→            />
   494→            {pwMismatch && <p className="text-xs text-rose-300">{t("settings.accountTab.pwMismatch")}</p>}
   495→          </div>
   496→        </div>
   497→        <div className="flex justify-end">
   498→          <Button
   499→            onClick={changePassword}
   500→            disabled={pwPending || !pwCurrent || !pwNew || !pwConfirm}
   501→            className="min-w-36"
   502→          >
   503→            {pwPending && <Loader2 className="h-4 w-4 animate-spin" />}
   504→            {t("settings.accountTab.updatePassword")}
   505→          </Button>
   506→        </div>
   507→      </section>
   508→
   509→      {/* Danger zone */}
   510→      <section className="rounded-xl border border-rose-500/25 bg-rose-500/[0.04] p-5">
   511→        <h3 className="font-display font-semibold text-rose-200">{t("settings.accountTab.dangerZone")}</h3>
   512→        <p className="mt-1 max-w-lg text-sm text-muted-foreground">
   513→          {t("settings.accountTab.dangerDescription")}
   514→        </p>
   515→        <AlertDialog open={delOpen} onOpenChange={closeDeleteDialog}>
   516→          <AlertDialogTrigger asChild>
   517→            <Button variant="destructive" className="mt-4">
   518→              <Trash2 className="h-4 w-4" />
   519→              {t("settings.accountTab.deleteAccount")}
   520→            </Button>
   521→          </AlertDialogTrigger>
   522→          <AlertDialogContent>
   523→            {delStage === "username" ? (
   524→              <>
   525→                <AlertDialogHeader>
   526→                  <AlertDialogTitle className="font-display">{t("settings.delete.title")}</AlertDialogTitle>
   527→                  <AlertDialogDescription>
   528→                    {t("settings.delete.typeUsername")}{" "}
   529→                    <span className="font-mono text-foreground">{user.username}</span>{" "}
   530→                    {t("settings.delete.toContinue")}
   531→                  </AlertDialogDescription>
   532→                </AlertDialogHeader>
   533→                <Input
   534→                  value={delConfirm}
   535→                  onChange={(e) => setDelConfirm(e.target.value)}
   536→                  placeholder={user.username}
   537→                  autoComplete="off"
   538→                  className="font-mono focus-glow"
   539→                  aria-label={t("settings.delete.usernameAria")}
   540→                />
   541→                <AlertDialogFooter>
   542→                  <AlertDialogCancel>{t("common.cancel")}</AlertDialogCancel>
   543→                  <AlertDialogAction
   544→                    disabled={delConfirm !== user.username}
   545→                    className="bg-rose-600 text-white hover:bg-rose-600/90"
   546→                    onClick={(e) => {
   547→                      e.preventDefault(); // keep the dialog open, advance to password step
   548→                      setDelStage("password");
   549→                    }}
   550→                  >
   551→                    {t("settings.delete.continue")}
   552→                  </AlertDialogAction>
   553→                </AlertDialogFooter>
   554→              </>
   555→            ) : (
   556→              <>
   557→                <AlertDialogHeader>
   558→                  <AlertDialogTitle className="font-display">
   559→                    {t("settings.delete.confirmTitle")}
   560→                  </AlertDialogTitle>
   561→                  <AlertDialogDescription>
   562→                    {t("settings.delete.confirmDescription")}
   563→                  </AlertDialogDescription>
   564→                </AlertDialogHeader>
   565→                <Input
   566→                  type="password"
   567→                  autoComplete="current-password"
   568→                  value={delPassword}
   569→                  onChange={(e) => {
   570→                    setDelPassword(e.target.value);
   571→                    setDelError(null);
   572→                  }}
   573→                  onKeyDown={(e) => {
   574→                    if (e.key === "Enter" && delPassword && !delPending) deleteAccount();
   575→                  }}
   576→                  placeholder={t("settings.delete.passwordPlaceholder")}
   577→                  className="focus-glow"
   578→                  aria-label={t("settings.delete.passwordAria")}
   579→                />
   580→                {delError && <p className="text-sm text-rose-300">{delError}</p>}
   581→                <AlertDialogFooter>
   582→                  <AlertDialogCancel onClick={() => setDelStage("username")}>
   583→                    {t("common.back")}
   584→                  </AlertDialogCancel>
   585→                  <AlertDialogAction
   586→                    disabled={delPending || !delPassword}
   587→                    className="bg-rose-600 text-white hover:bg-rose-600/90"
   588→                    onClick={(e) => {
   589→                      e.preventDefault(); // keep dialog open until the request succeeds
   590→                      deleteAccount();
   591→                    }}
   592→                  >
   593→                    {delPending ? (
   594→                      <Loader2 className="h-4 w-4 animate-spin" />
   595→                    ) : (
   596→                      <Trash2 className="h-4 w-4" />
   597→                    )}
   598→                    {t("settings.delete.deleteForever")}
   599→                  </AlertDialogAction>
   600→                </AlertDialogFooter>
   601→              </>
   602→            )}
   603→          </AlertDialogContent>
   604→        </AlertDialog>
   605→      </section>
   606→    </div>
   607→  );
   608→}

   610→// ─── NOTIFICATIONS ──────────────────────────────────────────────
   611→
   612→function NotificationsTab() {
   613→  const t = useT();
   614→  const [prefs, setPrefs] = useState<Prefs | null>(null);
   615→  const [pendingField, setPendingField] = useState<string | null>(null);
   616→
   617→  useEffect(() => {
   618→    let alive = true;
   619→    get<{ prefs: Prefs }>("/api/me/settings")
   620→      .then((res) => {
   621→        if (alive) setPrefs(res.prefs);
   622→      })
   623→      .catch((e) => {
   624→        toast.error(e instanceof Error ? e.message : t("settings.errors.loadPrefsFailed"));
   625→      });
   626→    return () => {
   627→      alive = false;
   628→    };
   629→  }, []);
   630→
   631→  async function togglePref(field: keyof Prefs, value: boolean) {
   632→    if (!prefs) return;
   633→    const previous = prefs;
   634→    setPrefs({ ...prefs, [field]: value }); // optimistic
   635→    setPendingField(field);
   636→    try {
   637→      const res = await patch<{ prefs: Prefs }>("/api/me/settings", { [field]: value });
   638→      setPrefs(res.prefs);
   639→      toast.success(t("settings.toasts.prefsUpdated"));
   640→    } catch (e) {
   641→      setPrefs(previous); // revert on error
   642→      toast.error(e instanceof Error ? e.message : t("settings.errors.prefsUpdateFailed"));
   643→    } finally {
   644→      setPendingField(null);
   645→    }
   646→  }
   647→
   648→  if (!prefs) {
   649→    return (
   650→      <div className="space-y-3">
   651→        {Array.from({ length: 3 }).map((_, i) => (
   652→          <Skeleton key={i} className="h-[76px] w-full rounded-xl bg-white/[0.05]" />
   653→        ))}
   654→      </div>
   655→    );
   656→  }
   657→
   658→  const rows: { field: keyof Prefs; title: string; description: string }[] = [
   659→    {
   660→      field: "projectActivity",
   661→      title: t("settings.notificationsTab.projectActivity.title"),
   662→      description: t("settings.notificationsTab.projectActivity.description"),
   663→    },
   664→    {
   665→      field: "collaborationRequests",
   666→      title: t("settings.notificationsTab.collaboration.title"),
   667→      description: t("settings.notificationsTab.collaboration.description"),
   668→    },
   669→    {
   670→      field: "socialActivity",
   671→      title: t("settings.notificationsTab.social.title"),
   672→      description: t("settings.notificationsTab.social.description"),
   673→    },
   674→  ];
   675→
   676→  return (
   677→    <div className="space-y-3">
   678→      {rows.map((row) => (
   679→        <div
   680→          key={row.field}
   681→          className="glass-card flex items-center justify-between gap-4 rounded-xl p-4"
   682→        >
   683→          <div className="min-w-0">
   684→            <p className="text-sm font-medium">{row.title}</p>
   685→            <p className="mt-0.5 text-xs leading-relaxed text-muted-foreground">
   686→              {row.description}
   687→            </p>
   688→          </div>
   689→          <Switch
   690→            checked={prefs[row.field]}
   691→            disabled={pendingField !== null}
   692→            onCheckedChange={(v) => togglePref(row.field, v)}
   693→            aria-label={t("settings.notificationsTab.toggleAria", { name: row.title })}
   694→          />
   695→        </div>
   696→      ))}
   697→    </div>
   698→  );
   699→}

   707→
   708→  const safetyItems: { icon: LucideIcon; title: string; text: string; tone: string }[] = [

   786→            </li>

   793→
   794→      <section className="glass-card rounded-xl p-5">
