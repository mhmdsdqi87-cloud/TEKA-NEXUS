   392→   392→export function SearchDialog() {
   393→   393→  const open = useApp((s) => s.searchOpen);
   394→   394→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   395→   395→  const navigate = useApp((s) => s.navigate);
   396→   396→
   397→   397→  const t = useT();
   398→   398→  const [query, setQuery] = useState("");
   399→   399→  const [data, setData] = useState<{ q: string; results: SearchResults } | null>(null);
   400→   400→  const reqId = useRef(0);
   401→   401→
   402→   402→  // Debounced server search (250ms, min 2 chars).
   403→   403→  useEffect(() => {
   404→   404→    const q = query.trim();
   405→   405→    if (q.length < 2) return;
   406→   406→    const id = ++reqId.current;
   407→   407→    const timer = setTimeout(async () => {
   408→   408→      try {
   409→   409→        const res = await get<SearchResults>(`/api/search${qs({ q })}`);
   410→   410→        if (reqId.current === id) setData({ q, results: res });
   411→   411→      } catch {
   412→   412→        if (reqId.current === id) setData({ q, results: { projects: [], users: [], tags: [] } });
   413→   413→      }
   414→   414→    }, 250);
   415→   415→    return () => clearTimeout(timer);
   416→   416→  }, [query]);
   417→   417→
   418→   418→  // Close + reset (event handler, not an effect).
   419→   419→  const handleOpenChange = (next: boolean) => {
   420→   420→    setSearchOpen(next);
   421→   421→    if (!next) {
   422→   422→      reqId.current++;
   423→   423→      setQuery("");
   424→   424→      setData(null);
   425→   425→    }
   426→   426→  };
   427→   427→
   428→   428→  const q = query.trim();
   429→   429→  // Results are only shown when they belong to the current query;
   430→   430→  // a newer query with no response yet renders the "Searching…" state.
   431→   431→  const results = data && data.q === q ? data.results : null;
   432→   432→  const searching = q.length >= 2 && results === null;
   433→   433→  const hasResults =
   434→   434→    results && (results.projects.length > 0 || results.users.length > 0 || results.tags.length > 0);
   435→   435→
   436→   436→  return (
   437→   437→    <CommandDialog
   438→   438→      open={open}
   439→   439→      onOpenChange={handleOpenChange}
   440→   440→      title={t("search.title")}
   441→   441→      description={t("search.description")}
   442→   442→      showCloseButton={false}
   443→   443→      className="max-w-xl top-[15%] translate-y-0"
   444→   444→    >
   445→   445→      <CommandInput
   446→   446→        value={query}
   447→   447→        onValueChange={setQuery}
   448→   448→        placeholder={t("search.placeholder")}
   449→   449→        aria-label={t("search.inputAria")}
   450→   450→      />
   451→   451→      <CommandList>

   675→   675→export function AppShell({ children }: { children: React.ReactNode }) {

   677→   677→  const setUnreadCount = useApp((s) => s.setUnreadCount);
   678→   678→  const setSearchOpen = useApp((s) => s.setSearchOpen);
   679→   679→  const t = useT();
   680→   680→  const userId = user?.id ?? null;
   681→   681→
   682→   682→  // Poll unread notification count every 45s while signed in.
   683→   683→  useEffect(() => {
   684→   684→    if (!userId) return;
   685→   685→    let cancelled = false;
   686→   686→    const poll = async () => {
   687→   687→      try {
   688→   688→        const res = await get<{ notifications: unknown[]; unread: number }>("/api/notifications");
   689→   689→        if (!cancelled) setUnreadCount(res.unread);
   690→   690→      } catch {
   691→   691→        /* silent — background badge poll */
   692→   692→      }
   693→   693→    };
   694→   694→    poll();
   695→   695→    const timer = setInterval(poll, 45_000);
   696→   696→    return () => {
   697→   697→      cancelled = true;
   698→   698→      clearInterval(timer);
   699→   699→    };
   700→   700→  }, [userId, setUnreadCount]);
   701→   701→
   702→   702→  // "/" opens global search (ignored while typing in form fields).
   703→   703→  useEffect(() => {
   704→   704→    const onKey = (e: KeyboardEvent) => {
   705→   705→      if (e.key !== "/") return;
   706→   706→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   707→   707→      const target = e.target as HTMLElement | null;
   708→   708→      const tag = target?.tagName;
   709→   709→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || target?.isContentEditable) return;
   710→   710→      e.preventDefault();
   711→   711→      setSearchOpen(true);
   712→   712→    };
   713→   713→    window.addEventListener("keydown", onKey);
   714→   714→    return () => window.removeEventListener("keydown", onKey);
   715→   715→  }, [setSearchOpen]);
   716→   716→
   717→   717→  return (
   718→   718→    <MotionConfig reducedMotion="user">
   719→   719→      <div className="min-h-screen flex">
   720→   720→        <LeftNav />
   721→   721→
   722→   722→        <div className="flex-1 flex flex-col min-w-0">
   723→   723→          <TopBar />
   724→   724→          <main className="flex-1 pb-24 md:pb-10" id="main-content">
   725→   725→            <div className="max-w-5xl mx-auto px-4 md:px-8 py-6">
   726→   726→              <motion.div
   727→   727→                initial={{ opacity: 0, y: 8 }}
   728→   728→                animate={{ opacity: 1, y: 0 }}
   729→   729→                transition={{ duration: 0.3, ease: "easeOut" }}
   730→   730→              >
   731→   731→                {children}
   732→   732→              </motion.div>
   733→   733→            </div>
   734→   734→          </main>
   735→   735→          <footer className="py-4 text-center text-xs text-muted-foreground/70 select-none" aria-label={t("nav.footerAria")}>
   736→   736→            {t("nav.footerTagline")}
   737→   737→          </footer>
   738→   738→        </div>

   741→   741→        <MobileNav />
   742→   742→        <SearchDialog />
