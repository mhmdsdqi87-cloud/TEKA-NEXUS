    90→    90→function useAcademyData(user: SessionUser | null) {
    91→    91→  const [progress, setProgress] = useState<ProgressMap>({});
    92→    92→  const [certs, setCerts] = useState<CertRow[]>([]);
    93→    93→  const [leaders, setLeaders] = useState<LeaderRow[] | null>(null);
    94→    94→  const [loaded, setLoaded] = useState(false);
    95→    95→
    96→    96→  const refresh = useCallback(async () => {
    97→    97→    if (!user) {
    98→    98→      setProgress({});
    99→    99→      setCerts([]);
   100→   100→      setLoaded(true);
   101→   101→      return;
   102→   102→    }
   103→   103→    try {
   104→   104→      const res = await get<{ progress: { courseId: string; completedLessons: string[]; lastLessonId: string | null }[]; certificates: CertRow[] }>("/api/academy");
   105→   105→      const map: ProgressMap = {};
   106→   106→      for (const p of res.progress) map[p.courseId] = { done: p.completedLessons, last: p.lastLessonId };
   107→   107→      setProgress(map);
   108→   108→      setCerts(res.certificates);
   109→   109→    } catch {
   110→   110→      /* offline — keep defaults */
   111→   111→    } finally {
   112→   112→      setLoaded(true);
   113→   113→    }
   114→   114→  }, [user]);
   115→   115→
   116→   116→  useEffect(() => {
   117→   117→    void refresh();
   118→   118→  }, [refresh]);
   119→   119→
   120→   120→  useEffect(() => {
   121→   121→    if (!user) {
   122→   122→      setLeaders(null);
   123→   123→      return;
   124→   124→    }
   125→   125→    get<{ rows: LeaderRow[] }>("/api/academy/leaderboard")
   126→   126→      .then((r) => setLeaders(r.rows))
   127→   127→      .catch(() => setLeaders([]));
   128→   128→  }, [user]);
   129→   129→
   130→   130→  return { progress, certs, leaders, loaded, refresh, setProgress, setCerts };
   131→   131→}

   135→   135→function ProgressRing({ value, size = 56 }: { value: number; size?: number }) {
   136→   136→  const stroke = 5;
   137→   137→  const r = (size - stroke) / 2;
   138→   138→  const c = 2 * Math.PI * r;
   139→   139→  return (
   140→   140→    <svg width={size} height={size} className="shrink-0 -rotate-90" role="img" aria-label={`${Math.round(value * 100)}%`}>
   141→   141→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.08)" strokeWidth={stroke} />
   142→   142→      <motion.circle
   143→   143→        cx={size / 2}
   144→   144→        cy={size / 2}
   145→   145→        r={r}
   146→   146→        fill="none"
   147→   147→        stroke="url(#ringGrad)"
   148→   148→        strokeWidth={stroke}
   149→   149→        strokeLinecap="round"
   150→   150→        strokeDasharray={c}
   151→   151→        initial={{ strokeDashoffset: c }}
   152→   152→        animate={{ strokeDashoffset: c * (1 - value) }}
   153→   153→        transition={{ duration: 1.1, ease: EASE }}
   154→   154→      />
   155→   155→      <defs>
   156→   156→        <linearGradient id="ringGrad" x1="0%" y1="0%" x2="100%" y2="100%">
   157→   157→          <stop offset="0%" stopColor="#22d3ee" />
   158→   158→          <stop offset="100%" stopColor="#a78bfa" />
   159→   159→        </linearGradient>
   160→   160→      </defs>
   161→   161→    </svg>
   162→   162→  );
   163→   163→}

   600→   600→function ModuleList({ course, doneSet, currentId, onNavigateLesson, t }: {
   601→   601→  course: Course;

   612→   612→      if (m.lessons.some((l) => l.id === currentId)) initial[m.id] = true;
   613→   613→    }

   619→   619→      {course.modules.map((m, mi) => {

   684→   684→export function AcademyView({ courseSlug, lessonId }: { courseSlug?: string; lessonId?: string }) {
   685→   685→  const t = useT();
   686→   686→  const user = useApp((s) => s.user);
   687→   687→  const navigate = useApp((s) => s.navigate);
   688→   688→  const setUser = useApp((s) => s.setUser);
   689→   689→  const { progress, certs, leaders, setProgress, refresh } = useAcademyData(user);
   690→   690→  const [certDialog, setCertDialog] = useState<CertInfo | null>(null);
   691→   691→  const course = useMemo(() => COURSES.find((c) => c.id === courseSlug), [courseSlug]);
   692→   692→
   693→   693→  const handleChallengePassed = useCallback(
   694→   694→    async (courseId: string, lessonId: string) => {
   695→   695→      if (!user) {
   696→   696→        toast.info(t("academy.loginPrompt"));
   697→   697→        return;
   698→   698→      }
   699→   699→      try {
   700→   700→        const res = await post<{
   701→   701→          added: boolean;
   702→   702→          xpAwarded: number;
   703→   703→          completedLessons: string[];
   704→   704→          courseCompleted: boolean;
   705→   705→          certificate: { code: string; createdAt: string } | null;
   706→   706→          user: { xp: number; reputation: number };
   707→   707→        }>("/api/academy/lessons", { courseId, lessonId });
   708→   708→
   709→   709→        setProgress((p) => ({
   710→   710→          ...p,
   711→   711→          [courseId]: { done: res.completedLessons, last: lessonId },
   712→   712→        }));
   713→   713→        setUser({ ...user, xp: res.user.xp, reputation: res.user.reputation });
   714→   714→
   715→   715→        if (res.added && res.xpAwarded > 0) {
   716→   716→          toast.success(t("academy.lessonDoneToast", { xp: res.xpAwarded }));
   717→   717→        }
   718→   718→        if (res.courseCompleted && res.certificate) {
   719→   719→          const c = COURSES.find((x) => x.id === courseId);
   720→   720→          toast.success(t("academy.courseDoneToast", { xp: res.xpAwarded }), { duration: 6000 });
   721→   721→          setCertDialog({
   722→   722→            courseTitle: c ? pick(c.title, "en") : courseId,
   723→   723→            recipient: user.displayName,
   724→   724→            code: res.certificate.code,
   725→   725→            issuedAt: new Date(res.certificate.createdAt).toLocaleDateString(),
   726→   726→            emoji: c?.emoji ?? "🏆",
   727→   727→          });
   728→   728→          void refresh();
   729→   729→        }
   730→   730→      } catch (e) {
   731→   731→        toast.error(e instanceof Error ? e.message : t("common.error"));
   732→   732→      }
   733→   733→    },
   734→   734→    [user, t, setProgress, setUser, refresh],
   735→   735→  );
   736→   736→
   737→   737→  const showCert = useCallback(
   738→   738→    (courseId: string, code: string, createdAt: string) => {
   739→   739→      const c = COURSES.find((x) => x.id === courseId);
   740→   740→      setCertDialog({
   741→   741→        courseTitle: c ? pick(c.title, "en") : courseId,
   742→   742→        recipient: user?.displayName ?? "TEKA Learner",
   743→   743→        code,
   744→   744→        issuedAt: new Date(createdAt).toLocaleDateString(),
   745→   745→        emoji: c?.emoji ?? "🏆",
   746→   746→      });
   747→   747→    },
   748→   748→    [user],
   749→   749→  );
   750→   750→
   751→   751→  return (
   752→   752→    <main className="mx-auto w-full max-w-6xl flex-1 px-4 py-6 sm:px-6 sm:py-8">
   753→   753→      {course ? (
   754→   754→        <CourseView
   755→   755→          course={course}
   756→   756→          initialLessonId={lessonId}
   757→   757→          progress={progress}
   758→   758→          onBack={() => navigate({ name: "academy" })}
   759→   759→          onNavigateLesson={(cid, lid) => navigate({ name: "academy", course: cid, lesson: lid })}
   760→   760→          onChallengePassed={handleChallengePassed}
   761→   761→          t={t}
   762→   762→        />
   763→   763→      ) : (
   764→   764→        <Hub
   765→   765→          onOpenCourse={(cid) => navigate({ name: "academy", course: cid })}
   766→   766→          progress={progress}
   767→   767→          certs={certs}
   768→   768→          leaders={leaders}
   769→   769→          onShowCert={showCert}
   770→   770→        />
   771→   771→      )}
   772→   772→
   773→   773→      <CertificateDialog
   774→   774→        cert={certDialog}
   775→   775→        open={!!certDialog}
   776→   776→        onClose={() => setCertDialog(null)}
   777→   777→        t={t}
   778→   778→      />
   779→   779→    </main>
   780→   780→  );
   781→   781→}
   782→   782→
